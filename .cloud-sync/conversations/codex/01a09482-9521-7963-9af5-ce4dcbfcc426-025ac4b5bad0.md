/**

async function listStudents(): Promise<StudentRow[]> {
  const { data, error } = await supabase.functions.invoke("admin-manage-users", {

/**
 * Resolve a name or email to exactly one student. Ambiguity is fatal rather
 * than resolved by picking the first: sending a student's homework to the wrong
 * account is silent — she simply never sees it, and he never learns why.
 */
function resolveStudent(rows: StudentRow[], query: string): StudentRow {
  const q = query.trim().toLowerCase();
  const exact = rows.filter(
    (r) => r.email?.toLowerCase() === q || r.display_name?.toLowerCase() === q,
  );
  const pool = exact.length ? exact : rows.filter(
    (r) =>
      r.display_name?.toLowerCase().includes(q) || r.email?.toLowerCase().includes(q),
  );

  if (pool.length === 0) {
    console.error(`assign: no student matches "${query}".`);
    console.error("Known:", rows.map((r) => r.display_name ?? r.email).join(" · "));
    process.exit(1);
  }
  if (pool.length > 1) {
    console.error(`assign: "${query}" is ambiguous — ${pool.length} accounts match:`);
    for (const r of pool) console.error(`  - ${r.display_name ?? "(no name)"}  <${r.email ?? "?"}>`);
    console.error("Narrow it with the full name or the email address.");
    process.exit(1);
  }
  return pool[0];
}

/* ── the week band ────────────────────────────────────────────────────── */

const ABBR = ["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"];

/** The Monday-based week containing `now`, as seven local ISO dates. */
/**
 * Seven days starting TODAY, not Monday to Sunday.
 *
 * A fixed calendar week hides tomorrow whenever today is Sunday, which is
 * exactly when a student most needs to see what is coming: work assigned for
 * Monday night simply was not on the strip. A rolling window always shows
 * today first and the next six days, and it re-labels itself as the days pass,
 * so tomorrow's assignment becomes today's without anything being rewritten.
 */
export function weekDates(now: Date = new Date()): string[] {
  const start = new Date(now);
  start.setHours(0, 0, 0, 0);
  return Array.from({ length: 7 }, (_, index) => {
    const day = new Date(start);
    day.setDate(start.getDate() + index);
    return localDayISO(day);
  });
}

export function weekLabelFor(dates: string[]): string {
  const first = parseLocalDay(dates[0]);
  const last = parseLocalDay(dates[dates.length - 1]);
  if (!first || !last) return "";
  const month = new Intl.DateTimeFormat("en-CA", { month: "long" });
  const same = first.getMonth() === last.getMonth();
  return same
    ? `${month.format(first)} ${first.getDate()} to ${last.getDate()}`
    : `${month.format(first)} ${first.getDate()} to ${month.format(last)} ${last.getDate()}`;
}

/**
 * Folds every dated thing in the week onto its day. `entries` are already
 * resolved to a local ISO day by the caller, so nothing is re-parsed here.
 */
export function buildWeek(
  dates: string[],
  entries: { iso: string; label: string }[],
  now: Date = new Date(),
): HomeWeekDay[] {
  const today = localDayISO(now);
  const byDay = new Map<string, string[]>();
  for (const entry of entries) {
    const bucket = byDay.get(entry.iso);
    if (bucket) bucket.push(entry.label);
    else byDay.set(entry.iso, [entry.label]);
  }

  return dates.map((iso) => {
    const date = parseLocalDay(iso);
    const labels = byDay.get(iso) ?? [];
    const weekday = date ? date.getDay() : 1;
    return {
      iso,
      abbr: ABBR[weekday],
      dayNum: date ? date.getDate() : 0,
      isToday: iso === today,
      isWeekend: weekday === 0 || weekday === 6,
      // One line per cell: the first thing, and a count when there is more.
      label:
        labels.length === 0
          ? null
          : labels.length === 1
            ? labels[0]
            : `${labels[0]} +${labels.length - 1}`,
    };
  });
}

export function isSelfContainedItem(item: ClassroomItem): boolean {
  return !item.source_id && !!(item.content_html || item.content_markdown);
}

/**
 * Resolve a classroom_items row to the right student-facing route.
 * Returns the path to navigate to, or null if the item is not actionable.
 */
export function resolveItemRoute(item: ClassroomItem): string | null {
  switch (item.type) {
    case 'drill':
      if (item.source_table === 'guided_drill_assignments' && item.source_id) {
        return `/guided/${encodeURIComponent(item.source_id)}?returnTo=${encodeURIComponent('/classroom')}`;
      }
      if (item.source_id) return `/drill?assignmentId=${encodeURIComponent(item.source_id)}`;
      return '/drill';
    case 'homework':
      if (item.source_id) return `/classroom/${encodeURIComponent(item.source_id)}`;
      if (isSelfContainedItem(item)) return `/classroom/item/${encodeURIComponent(item.id)}`;
      return '/classroom';
    case 'activity':
      // source_id is the activity_assignments row id; source_table='activity_assignments'.
      if (item.source_id) return `/activities/${encodeURIComponent(item.source_id)}`;
      return '/activities';
    case 'bootcamp_module':
      if (item.source_id) return item.source_id; // expected to be a /bootcamp/* path
      return '/bootcamps';
    case 'session_recap':
    case 'smartboard_recap':
    case 'announcement':
    case 'note':
      return `/classroom/item/${encodeURIComponent(item.id)}`;
    default:
      return `/classroom/item/${encodeURIComponent(item.id)}`;
  }
}
