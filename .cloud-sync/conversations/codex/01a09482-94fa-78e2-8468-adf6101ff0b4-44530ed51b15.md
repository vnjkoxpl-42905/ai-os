/** Short, stable, filename-safe digest of whatever identifies the run. */
function fingerprint(input: string): string {
  let h = 2166136261;
  for (let i = 0; i < input.length; i++) {
    h ^= input.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return (h >>> 0).toString(36);
}

/**
 * Key one run's resume row.
 *
 * A full section is identified by the section itself; a filtered drill by the
 * filters that drew it; Smart Drill by nothing but its mode, because it has a
 * single continuous run per student. Returns null for the surfaces that own
 * their own resume rows (guided, practice sets) and for anything launched from
 * an explicit list of qids, where the caller already knows the set.
 */
export function drillResumeKey(input: {
  mode: string;
  config?: unknown;
  identity?: DrillResumeIdentity;
}): string | null {
  if (input.identity?.kind === 'bootcamp-day') {
    return `${DRILL_SESSION_PREFIX}bootcamp-day::${input.identity.day}::${input.identity.part}`;
  }

  const mode = input.mode;
  if (mode === 'adaptive') return `${DRILL_SESSION_PREFIX}adaptive`;

  const config = (input.config ?? null) as Record<string, unknown> | null;

  if (mode === 'full-section') {
    if (!config) return null;
    const pt = config.pt;
    const section = config.section;
    if (pt == null || section == null) return null;
    return `${DRILL_SESSION_PREFIX}section::pt${pt}-s${section}`;
  }

  if (mode === 'type-drill' || mode === 'natural-drill') {
    if (!config) return null;
    const selected = config.selectedQids as string[] | undefined;
    if (selected?.length) {
      // A pre-selected set is its own identity: the same list re-opened is the
      // same run, a different list is a different one.
      return `${DRILL_SESSION_PREFIX}${mode}::set-${fingerprint([...selected].sort().join(','))}`;
    }
    const shape = JSON.stringify({
      qtypes: config.qtypes ?? [],
      difficulties: config.difficulties ?? [],
      pts: config.pts ?? [],
      sections: config.sections ?? [],
      qfeatures: config.qfeatures ?? [],
      reasoningTypes: config.reasoningTypes ?? [],
      searchQuery: config.searchQuery ?? '',
      questionNumber: config.questionNumber ?? null,
      count: config.count ?? null,
    });
    return `${DRILL_SESSION_PREFIX}${mode}::${fingerprint(shape)}`;
  }

  return null;
}

/**
 * Deterministic per-user, per-pool record id, so every read and write lands on
 * the same row without a lookup. Mirrors progressIdFor's FNV-1a approach but
 * hashes user + slug together.
 */
export function practiceSessionIdFor(userId: string, poolSlug: string): string {
  const input = `${userId}::${poolSlug}`;
  let h = 0x811c9dc5;
  for (let i = 0; i < input.length; i++) {
    h ^= input.charCodeAt(i);
    h = Math.imul(h, 0x01000193);
  }
  let h2 = 0x811c9dc5;
  for (let i = input.length - 1; i >= 0; i--) {
    h2 ^= input.charCodeAt(i);
    h2 = Math.imul(h2, 0x01000193);
  }
  const hex = (h >>> 0).toString(16).padStart(8, '0');
  const hex2 = (h2 >>> 0).toString(16).padStart(8, '0');
  return `00000000-0000-4000-8000-${hex}${hex2.slice(0, 4)}`;
}

/**
 * Read the stored session for one pool. Returns null when there is nothing to
 * resume, when the row is corrupt, or when persistence is unavailable — all of
 * which mean "start fresh", never "block the student".
 */
export async function loadPracticeSession(
  userId: string,
  poolSlug: string,
): Promise<PracticeSession | null> {
  if (!userId || !poolSlug) return null;
  try {
    const persist = getPersistence(userId);
    // Wait for cloud hydration first, otherwise a cold load on a second device
    // reports "nothing to resume" while the row is still in flight.
    await persist.ready();
    const raw = await persist.get<unknown>(TABLES.practice_sessions, practiceSessionIdFor(userId, poolSlug));
    if (!raw) return null;

    if (!parsed.success) {
      console.warn('[practice-session] stored row failed validation, ignoring', {
        poolSlug,
        issues: parsed.error.issues,
      });
      return null;
    }
    return parsed.data;
  } catch (e) {
    console.warn('[practice-session] load failed, starting fresh', { poolSlug, error: e });
    return null;
  }
}

/**
 * The clock a full section was started under, in milliseconds. Unlimited has
 * no budget, which is not the same as a budget of zero.
 */
function sectionBudgetMs(timer?: TimerConfig): number | undefined {
  if (!timer) return undefined;
  switch (timer.mode) {
    case '35':
      return 35 * 60 * 1000;
    case '52.5':
      return 52.5 * 60 * 1000;
    case '70':
      return 70 * 60 * 1000;
    case 'custom':
      return timer.customMinutes ? timer.customMinutes * 60 * 1000 : undefined;
    default:
      return undefined;
  }
}
