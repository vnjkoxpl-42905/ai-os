177	function practiceDiagrams(item: PracticeItem) {
178	  const left = `${item.leftNegated ? '~' : ''}${item.leftCode}`;
179	  const right = `${item.rightNegated ? '~' : ''}${item.rightCode}`;
180	  const notLeft = `${item.leftNegated ? '' : '~'}${item.leftCode}`;
181	  const notRight = `${item.rightNegated ? '' : '~'}${item.rightCode}`;
182	  return { original: `${left} → ${right}`, contra: `${notRight} → ${notLeft}`, reverse: `${right} → ${left}`, inverse: `${notLeft} → ${notRight}` };
183	}
184	function practiceChoices(item: PracticeItem, contrapositive: boolean) {
185	  const diagrams = practiceDiagrams(item);
186	  const correct = contrapositive ? diagrams.contra : diagrams.original;
187	  const wrong = [diagrams.reverse, diagrams.inverse];
188	  const position = contrapositive ? item.contraPosition : item.originalPosition;
189	  const options = [...wrong];
190	  options.splice(position, 0, correct);
191	  return options;
192	}
