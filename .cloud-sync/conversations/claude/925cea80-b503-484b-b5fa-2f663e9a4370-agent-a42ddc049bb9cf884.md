    82	const QID_PATTERN = /^PT(\d+)-S(\d+)-Q\d+$/;

    84	/**
    85	 * A homework set built from one real section — every qid comes from the same
    86	 * PT and section number, the shape `assign.ts` and the admin set builder both
    87	 * produce today. Such a set is a section wearing a homework row, not a mixed
    88	 * question list, so it runs under a real countdown instead of the untimed
    89	 * stopwatch every other homework set gets. A set that mixes sections (or PTs)
    90	 * returns null and keeps the untimed behaviour, because there is no one clock
    91	 * that honestly times a blend.
    92	 */
    93	export function singleSectionOf(qids: string[]): { pt: number; section: number } | null {
    94	  if (qids.length === 0) return null;
    95	  let pt: number | null = null;
    96	  let section: number | null = null;
    97	  for (const qid of qids) {
    98	    const match = QID_PATTERN.exec(qid);
    99	    if (!match) return null;
   100	    const qPt = Number(match[1]);
   101	    const qSection = Number(match[2]);
   102	    if (pt === null) {
   103	      pt = qPt;
   104	      section = qSection;
   105	    } else if (qPt !== pt || qSection !== section) {
   106	      return null;
   107	    }
   108	  }
   109	  return pt !== null && section !== null ? { pt, section } : null;
   110	}

   266	export function timerConfigForRunSpec(
   267	  spec: DrillRunSpec | null | undefined,
   268	): DrillTimerConfig | null {
   269	  if (spec?.isGuidedDrill && spec.guidedDurationMinutes) {
   270	    return { mode: 'countdown', durationMs: spec.guidedDurationMinutes * 60 * 1000 };
   271	  }
   272	
   273	  if ((spec?.mode === 'type-drill' || spec?.mode === 'practice-set') && spec.config) {
   274	    const config = spec.config as TypeDrillConfig;
   275	    if (config.selectedQids && config.selectedQids.length > 0) {
   276	      return { mode: 'stopwatch' };
   277	    }
   278	  }
   279	
   280	  if (spec?.mode === 'full-section' && spec.config) {
   281	    const timer = (spec.config as FullSectionConfig).timer as TimerConfig;
   282	
   283	    if (timer.mode === 'unlimited') {
   284	      return { mode: 'stopwatch' };
   285	    }
   286	
   287	    let durationMs: number;
   288	    switch (timer.mode) {
   289	      case '35':
   290	        durationMs = 35 * 60 * 1000;
   291	        break;
   292	      case '52.5':
   293	        durationMs = 52.5 * 60 * 1000;
   294	        break;
   295	      case '70':
   296	        durationMs = 70 * 60 * 1000;
   297	        break;
   298	      case 'custom':
   299	        durationMs = (timer.customMinutes || 35) * 60 * 1000;
   300	        break;
   301	      default:
   302	        durationMs = 35 * 60 * 1000;
   303	    }
   304	
   305	    return { mode: 'countdown', durationMs };
   306	  }
   307	
   308	  return { mode: 'stopwatch' };

  if (!metadata || typeof metadata !== "object") return null;

    minutes,
    sectionName: SECTION_NAMES[sectionKind] ?? "Timed Section",
    href,
    sealed: meta.sealed === true,
  };
