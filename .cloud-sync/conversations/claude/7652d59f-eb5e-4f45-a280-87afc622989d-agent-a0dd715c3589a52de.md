     6	/**
     7	 * The mouse-up ladder for one markable surface: the passage, a question stem,
     8	 * or an answer choice. Every surface runs the same ladder so the toolbar tools,
     9	 * the eraser and the selection popover behave identically wherever the student
    10	 * releases the mouse.
    11	 *
    12	 *   erase          -> drag clears the sweep, click clears the mark under it
    13	 *   tool active    -> a selection is applied on release, no popover
    14	 *   click on mark  -> "Remove highlight" popover
    15	 *   selection      -> "Highlight" popover
    16	 *
    17	 * The popover itself belongs to the pane, since a pane with several surfaces
    18	 * (a stem and five choices) shows one popover. `onAnchor` hands the pane the
    19	 * position together with a handle back to the surface that asked, so the
    20	 * popover's buttons act on the right text.
    21	 */
    22	
    23	export interface RcMarkSurfaceHandle {
    24	  /** Popover "Highlight": paint the selection the last mouse-up captured. */
    25	  applyPending: (chosen: RcHighlightColor) => void;
    26	  /** Popover "Remove highlight": drop the mark the last mouse-up landed on. */
    27	  removePending: () => void;
    28	}

    52	  // The handle is created once and read by the pane's popover later, so it
    53	  // reads the options through a ref rather than closing over one render's.
    54	  const latest = React.useRef(options);
    55	  latest.current = options;
    56	
    57	  const pending = React.useRef<{ start: number; end: number } | null>(null);
    58	  const removeOffset = React.useRef<number | null>(null);
    59	
    60	  const handle = React.useMemo<RcMarkSurfaceHandle>(() => {
    61	    const self: RcMarkSurfaceHandle = {
    62	      applyPending: (chosen) => {
    63	        if (pending.current) {
    64	          latest.current.onAdd(pending.current.start, pending.current.end, chosen, 'highlight');
    65	        }
    66	        window.getSelection()?.removeAllRanges();
    67	        latest.current.onAnchor(null, self);
    68	      },
    69	      removePending: () => {
    70	        if (removeOffset.current != null) latest.current.onRemove(removeOffset.current);
    71	        latest.current.onAnchor(null, self);
    72	      },
    73	    };
    74	    return self;
    75	  }, []);
    76

    78	   * Popover position, in the pane's content coordinates. The pane scrolls and
    79	   * the popover is positioned inside it, so the scroll offset is added back:
    80	   * client rects are measured against the viewport, not the scrolled content.
    81	   */

    83	    const pane = latest.current.paneRef.current!;

    86	      x: rect.left - paneRect.left + pane.scrollLeft,
    87	      y: rect.bottom - paneRect.top + 6 + pane.scrollTop,
    88	      mode,
    89	    };
    90	  };

   115	      }
   116	      onAnchor(null, handle);
   117	      return;
   118	    }
