function sectionBudgetMs(timer?: TimerConfig): number | undefined {
  if (!timer) return undefined;
  switch (timer.mode) {
    case '35':
      return 35 * 60 * 1000;
    case '52.5':
      return 52.5 * 60 * 1000;
    case '70':
      return 70 * 60 * 1000;
    case 'custom':
      return timer.customMinutes ? timer.customMinutes * 60 * 1000 : undefined;
    default:
      return undefined;
  }
}

   153	/**
   154	 * Totals over the per-question stats.
   155	 *
   156	 * Guards the division: a section with no resolvable questions used to produce
   157	 * `NaN` for both percent and average, and Postgres rejects `NaN` on a numeric
   158	 * column, so the whole insert failed on an edge case that should simply record
   159	 * zeroes.
   160	 */
   161	/**
   162	 * Was a blind review actually done? One predicate, used by both the totals and
   163	 * the persisted row, so the score on screen and the row in section_history can
   164	 * never disagree about whether a review happened.
   165	 */
   166	const hasBlindReview = (brResults?: BlindReviewResult[]): boolean =>
   167	  Array.isArray(brResults) && brResults.length > 0;

   169	export function buildSectionTotals(
   170	  stats: SectionQuestionStat[],
   171	  brResults?: BlindReviewResult[],
   172	): SectionTotals {
   173	  const initialTotal = stats.length;
   174	  const initialScore = stats.filter((s) => s.initialCorrect).length;
   175	  const initialPercent = initialTotal
   176	    ? Math.round((initialScore / initialTotal) * 100)
   177	    : 0;
   178	
   179	  // An EMPTY array is not a blind review. `brResults ?` treated `[]` as one,
   180	  // so a student who skipped straight to the score report was told they had
   181	  // scored 0 on a review they never sat, with a delta to match. Null is the
   182	  // only honest answer when there is nothing to report.
   183	  const brScore = hasBlindReview(brResults) ? stats.filter((s) => s.brCorrect).length : null;
   184	  const brPercent =
   185	    brScore !== null && initialTotal
   186	      ? Math.round((brScore / initialTotal) * 100)
   187	      : null;
   188	  const delta = brScore !== null ? brScore - initialScore : null;
   189	
   190	  const totalTimeMs = stats.reduce((sum, s) => sum + s.timeMs, 0);
   191	  const avgTimeMs = initialTotal ? Math.round(totalTimeMs / initialTotal) : 0;
   192	  const unansweredCount = stats.filter(
   193	    (s) => s.initialAnswer === UNANSWERED,
   194	  ).length;
   195	
   196	  return {
   197	    initialScore,
   198	    initialTotal,
   199	    initialPercent,
   200	    brScore,
   201	    brPercent,
   202	    delta,
   203	    totalTimeMs,
   204	    avgTimeMs,
   205	    unansweredCount,
   206	  };
   207	}

   209	/** A `{ correct, total }` tally keyed however the caller groups the stats. */
   210	function tally<K extends string | number>(
   211	  stats: SectionQuestionStat[],
   212	  key: (s: SectionQuestionStat) => K,
   213	): Record<K, { correct: number; total: number }> {
   214	  const out = {} as Record<K, { correct: number; total: number }>;
   215	  stats.forEach((s) => {
   216	    const k = key(s);
   217	    if (!out[k]) out[k] = { correct: 0, total: 0 };
   218	    out[k].total++;
   219	    if (s.initialCorrect) out[k].correct++;
   220	  });
   221	  return out;
   222	}

-- Immutability guard: students cannot mutate snapshot or attribution columns.
-- Allowed student writes: status, started_at, completed_at, score.
CREATE OR REPLACE FUNCTION public.guard_homework_assignment_immutables()
RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF public.has_role(auth.uid(), 'admin') THEN
    RETURN NEW;
  END IF;

  IF NEW.set_id           IS DISTINCT FROM OLD.set_id           OR
     NEW.student_id       IS DISTINCT FROM OLD.student_id       OR
     NEW.question_qids    IS DISTINCT FROM OLD.question_qids    OR
     NEW.set_title        IS DISTINCT FROM OLD.set_title        OR
     NEW.set_description  IS DISTINCT FROM OLD.set_description  OR

     NEW.created_at       IS DISTINCT FROM OLD.created_at THEN
    RAISE EXCEPTION 'homework_assignments: snapshot columns are immutable for non-admin users';
  END IF;

  RETURN NEW;
END;
