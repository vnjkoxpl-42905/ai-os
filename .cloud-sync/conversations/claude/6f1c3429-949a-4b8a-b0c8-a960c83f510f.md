export interface QueueEntry { id: string; retry: boolean }
/** Everything the deck needs to resume exactly where the student stopped. */
export interface DeckState {
  v: 1;
  round: number;
  queue: QueueEntry[];
  index: number;
  /** The group the student chose on the current card, until they move on. */

const byId = new Map(indicatorWordCards.map(card => [card.id, card]));

export function mulberry32(seed: number) {
  return () => {
    seed = (seed + 0x6D2B79F5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function shuffled<T>(list: T[], rng: () => number): T[] {
  const copy = [...list];
  for (let i = copy.length - 1; i > 0; i -= 1) {
    const j = Math.floor(rng() * (i + 1));
    [copy[i], copy[j]] = [copy[j], copy[i]];
  }
  return copy;
}
export function longestRun(cards: IndicatorWordCard[]): number {
  let best = 0; let run = 0;
  cards.forEach((card, i) => { run = i && cards[i - 1].group === card.group ? run + 1 : 1; best = Math.max(best, run); });
  return best;
}
const triples = (cards: IndicatorWordCard[]) => cards.reduce((n, card, i) => n + (i >= 2 && cards[i - 1].group === card.group && cards[i - 2].group === card.group ? 1 : 0), 0);
/** A true shuffle, then random swaps until no group repeats more than twice in a row. Some back-to-backs stay; long streaks do not. */
export function shuffleSpread(cards: IndicatorWordCard[], rng: () => number = Math.random): IndicatorWordCard[] {
  const order = shuffled(cards, rng);
  let bad = triples(order);
  for (let step = 0; step < 4000 && bad > 0; step += 1) {
    const i = Math.floor(rng() * order.length);
    const j = Math.floor(rng() * order.length);
    if (i === j || order[i].group === order[j].group) continue;
    [order[i], order[j]] = [order[j], order[i]];
    const now = triples(order);
    if (now <= bad) bad = now; else [order[i], order[j]] = [order[j], order[i]];
  }
  return order;
}

export function newRound(previous?: DeckState, rng: () => number = Math.random): DeckState {
  return {
    v: 1, round: previous ? previous.round + 1 : 1,
    queue: shuffleSpread(indicatorWordCards, rng).map(card => ({ id: card.id, retry: false })),
    index: 0, picked: null, results: {},
    earlier: previous ? [...previous.earlier, { right: summarize(previous).right, total: indicatorWordCards.length }] : [],
  };
}

export const isFinished = (state: DeckState) => state.index >= state.queue.length;

/** Record the student's choice. A first-pass miss queues that card once, at the very end of the round. */
export function pick(state: DeckState, choice: IndicatorGroup): DeckState {
  const entry = state.queue[state.index];
  const card = entry && cardById(entry.id);
  if (!card || state.picked !== null) return state;
  const outcome: Outcome = choice === card.group ? 'right' : 'wrong';
  const prior = state.results[card.id] ?? {};
  const results = { ...state.results, [card.id]: entry.retry ? { ...prior, second: outcome } : { ...prior, first: outcome } };
  const queue = outcome === 'wrong' && !entry.retry ? [...state.queue, { id: card.id, retry: true }] : state.queue;
  return { ...state, picked: choice, results, queue };
}
export function advance(state: DeckState): DeckState {
  return state.picked === null ? state : { ...state, index: state.index + 1, picked: null };
}

export function summarize(state: DeckState) {
  const firstTotal = state.queue.filter(entry => !entry.retry).length;
  const entries = Object.entries(state.results);
  const right = entries.filter(([, r]) => r.first === 'right').length;
  const missed = entries.filter(([, r]) => r.first === 'wrong').length;
  const stillMissing = entries.filter(([, r]) => r.first === 'wrong' && r.second === 'wrong').map(([id]) => cardById(id)!.word);
  return { right, missed, firstTotal, stillMissing };
}

/** Accepts only a state that still matches the deck; anything else starts a fresh round rather than resuming into a broken one. */
export function readDeckState(raw: unknown): DeckState | null {
  const value = raw as Partial<DeckState> | null;
  if (!value || typeof value !== 'object' || value.v !== 1 || !Number.isInteger(value.round) || !Number.isInteger(value.index)) return null;
  if (!Array.isArray(value.queue) || !value.queue.every(entry => entry && typeof entry.id === 'string' && byId.has(entry.id) && typeof entry.retry === 'boolean')) return null;
  const firstPass = value.queue.filter(entry => !entry.retry);
  if (firstPass.length !== indicatorWordCards.length || new Set(firstPass.map(entry => entry.id)).size !== indicatorWordCards.length) return null;
  if (value.index! < 0 || value.index! > value.queue.length) return null;
  if (value.picked !== null && value.picked !== 1 && value.picked !== 2) return null;
  if (value.picked !== null && value.index === value.queue.length) return null;
  if (!value.results || typeof value.results !== 'object') return null;
  const outcome = (v: unknown) => v === undefined || v === 'right' || v === 'wrong';
  if (!Object.entries(value.results).every(([id, r]) => byId.has(id) && !!r && outcome(r.first) && outcome(r.second))) return null;
  const earlier = Array.isArray(value.earlier) ? value.earlier.filter(r => r && Number.isInteger(r.right) && Number.isInteger(r.total)) : [];
  return { v: 1, round: value.round!, queue: value.queue, index: value.index!, picked: value.picked ?? null, results: value.results, earlier };
}

# Grill Me

One skill, self-contained. It holds both the **interview method** (Step 2) and the **capture discipline** (Steps 1 and 3): every answer lands in a durable markdown file, so nothing is lost as context fills up. It does not call the vendor `grilling` skill: the method is written inline below so this skill can add the capture steps around it and keep working even if `grilling` changes upstream.

If the topic is a feature, component, or other implementation work rather than an open-ended plan or decision, `brainstorming` runs a similar interview but gates on a written, approved design doc before any code — reach for that instead when the destination is implementation.

## Step 1: set up the capture file (BEFORE the first question)

1. Read the applicable `AGENTS.md` and `CLAUDE.md`, then only the context pages and prior captures relevant to the topic. Starter-kit context lives in `context/`; in an established project follow its declared routes instead of creating a duplicate knowledge store. Reuse facts already supplied; ask about gaps, changes, decisions, and tradeoffs. Installing the skill does not start an interview.
2. Create the capture file at `brainstorms/{YYYY-MM-DD}-{topic-slug}.md` (create `brainstorms/` if missing). Every capture lives here, one predictable home regardless of topic. Do NOT scatter captures into project folders. A polished deliverable (plan, map, spec) may later move into the relevant `projects/` folder; the raw capture always stays in `brainstorms/`.
   - Get today's real date from the clock or shell (`Get-Date -Format yyyy-MM-dd` in PowerShell, `date +%F` in Bash).
   - Never overwrite an existing capture. Same date and topic: add a time or unique suffix and create exclusively. Resume an existing capture only when the user asks or clearly refers to that session; read it first and continue its question numbering.
   - No topic supplied or inferable: create an `untitled-discovery` capture with goal pending, ask what to explore, record the chosen topic and goal in that same file.
3. Write the header immediately: title, date, goal of the session, empty "Open flags" section.
4. Tell the user where you are saving, in one line.

## Step 2: run the interview

Interview the user relentlessly until you reach a shared understanding. Map this as a **design tree**: every decision branches into the decisions that hang off it.

Work the tree in **rounds**. The **frontier** is every decision whose prerequisites are already settled: the questions you can ask _now_ without guessing at answers you have not heard yet. Ask the whole frontier in one round: number each question and give your recommended answer. Then wait for the user's answers before the next round.

Format a round like so:

```
**Q1 - <question title>**: <question body, may be multiple paragraphs, including choices>

Recommended: <your recommended answer>

---

**Q2 - <question title>**: ...

Recommended: ...
```

Each round the user answers reshapes the tree: settled decisions push the frontier outward and unblock questions that depended on them. Recompute the frontier and ask the next round. A question whose answer depends on another question still open in this round belongs to a _later_ round, not this one.

Finding _facts_ is your job, never the user's. When a frontier question needs a fact from the environment (filesystem, tools, live site), look it up or dispatch a sub-agent; never ask the user for anything you could find yourself. Do not block on it: a running exploration is an unsettled prerequisite, so only the questions downstream of it wait; ask the rest of the frontier now. The _decisions_ are the user's: put each to them and wait.

The session is done when the frontier is empty: every branch visited, nothing silently assumed. Do not act on it until the user confirms you have reached a shared understanding.

Two adjustments the capture layer imposes on the method:

- For **decisions**, give a recommended answer grounded in context and label it as a recommendation. For **personal or business facts**, ask neutrally. Never lead with invented facts, and never save your inference as the user's answer.
- When the user **cannot answer** something, capture it as a flag with the right owner and move on. Do not stall the round on it.

## Step 3: the checkpoint rule (non-negotiable)

After EVERY round of answers, BEFORE you ask the next round:

- Append one structured entry per answered question: the question, the key facts and decisions from the answer (in the user's words where wording matters), and any flags (things they could not answer, plus who should).

- Read back the saved entries before continuing. If the write fails, say so, keep the answers visible in chat, and do not claim they were saved or collect more answers without a working checkpoint.
- Only then ask the next round.

Never let two rounds pile up in your head before writing. The file, not your context, is the source of truth. Never make the user ask you to save progress.

Keep user-confirmed facts, tentative ideas, your recommendations, and unresolved questions distinguishable in the capture.
