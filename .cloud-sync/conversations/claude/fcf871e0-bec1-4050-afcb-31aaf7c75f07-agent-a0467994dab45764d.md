410	    { start: indicatorStart, end: indicatorStart + item.word.length, kind: 'indicator' },
411	    { start: targetStart, end: targetStart + item.sentenceTarget.length, kind: 'target' },
412	  ].sort((a, b) => a.start - b.start);

416	    if (!marked) { setConnector(null); return; }

418	      if (!figure.current || !sentence.current || !indicator.current || !target.current) return;
419	      const origin = figure.current.getBoundingClientRect();

421	      const targetRects = target.current.getClientRects();

428	      // Starting under an early line could otherwise cross a later wrapped clause.

435	    if (figure.current) observer?.observe(figure.current);
