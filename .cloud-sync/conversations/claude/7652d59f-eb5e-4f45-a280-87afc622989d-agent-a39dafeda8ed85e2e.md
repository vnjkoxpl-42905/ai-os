/**
 * One answer choice, using the LR player's row markup and state ladder so the
 * two drills are indistinguishable at the choice level.
 *
 * Two deliberate departures from LR, both taken from the official RC interface:
 * the letter badge is a circle with a visible outline rather than a rounded
 * square with a transparent border, and the eliminate control is always visible
 * rather than appearing on row hover.
 *
 * Like LR, a revealed question greens the credited answer and leaves the
 * student's pick in the selected slate — there is no red row. The verdict line
 * and the "Your answer" chip carry that.
 *
 * The credited answer's colours come from `.rc-choice-correct` in rc.css rather
 * than Tailwind utilities. They used to be green-500/green-50 here while the
 * verdict banner directly above used the design system's drill-correct #2f9e64
 * on #eaf6ef — two greens for one verdict, on one screen. Both read the same
 * token now, and it is the documented one.
 *
 * The choice text is a markable surface, like the passage: the student can
 * highlight and underline inside it with the same tools. That is why the text
 * is not a button. Chromium will not start a drag-selection inside one, and a
 * disabled button swallows the mouse entirely, which would make a revealed or
 * crossed-out choice impossible to mark. A plain click on the text still
 * answers; the letter badge stays the keyboard and screen-reader control.
 */

interface Props {
  choice: RcChoice;
  /** Scopes this choice's marks: highlights are stored per passage. */
  questionId: number;
  selected: boolean;
  eliminated: boolean;
  /** Set once feedback is showing, so the row can carry the verdict colors. */
  revealed: boolean;
  onSelect: (letter: RcLetter) => void;
  onToggleElimination: (letter: RcLetter) => void;
  disabled?: boolean;
  /**
   * Disables the cross-out control ALONE, leaving the choice selectable.
   *
   * Kept separate from `disabled` on purpose. A blind re-attempt has to hide
   * the student's original cross-outs while they answer again, and it also has
   * to let them answer — so the row cannot simply be disabled. Before this
   * existed, the eye stayed live while its effect was suppressed: the click
   * reached the reducer, the row never changed, and the label never flipped to
   * "Restore". A control that does nothing is worse than one that is off.
   */
  eliminationDisabled?: boolean;
  textStyle?: React.CSSProperties;
  badgeStyle?: React.CSSProperties;
  /** The per-choice disclosure, rendered under the text in review. */
  children?: React.ReactNode;
  /** This passage's marks. The row paints only the ones on its own text. */
  highlights: RcHighlight[];
  color: RcHighlightColor;
  highlightMode: RcHighlightMode;
  onAddHighlight: (
    target: RcHighlightTarget,
    start: number,
    end: number,
    color: RcHighlightColor,
    kind: RcAnnotationKind,
  ) => void;
  onRemoveHighlight: (target: RcHighlightTarget, offset: number) => void;
  onRemoveRange: (target: RcHighlightTarget, start: number, end: number) => void;
  /** The pane owns the one popover; the row reports where it should sit. */
  onAnchor: (anchor: RcSelectionAnchor | null, surface: RcMarkSurfaceHandle) => void;
  paneRef: React.RefObject<HTMLElement | null>;
}

export function RcChoiceRow({
  choice,
  questionId,
  selected,
  eliminated,
  revealed,
  onSelect,
  onToggleElimination,
  disabled,
  eliminationDisabled,
  textStyle,
  badgeStyle,
  children,
  highlights,
  color,
  highlightMode,
  onAddHighlight,
  onRemoveHighlight,
  onRemoveRange,
  onAnchor,
  paneRef,
}: Props) {
  const isCorrect = revealed && choice.correct;
  const noCrossOut = disabled || eliminationDisabled;
  const pickable = !disabled && !eliminated;
  const pick = () => {
    if (pickable) onSelect(choice.letter);
  };

  const textRef = React.useRef<HTMLDivElement>(null);
  const textId = React.useId();
  const target = questionTarget(questionId, choice.letter);
  const choiceHighlights = React.useMemo(
    () => highlights.filter((h) => h.target === target),
    [highlights, target],
  );

  React.useLayoutEffect(() => {
    const el = textRef.current;
    if (el) paintRcAnnotations(el, choiceHighlights);
  }, [choice.letter, choice.text, choiceHighlights]);

  const surface = useRcMarkSurface({
    textRef,
    paneRef,
    highlightMode,
    color,
    onAdd: (start, end, chosen, kind) => onAddHighlight(target, start, end, chosen, kind),
    onRemove: (offset) => onRemoveHighlight(target, offset),
    onRemoveRange: (start, end) => onRemoveRange(target, start, end),
    onAnchor,
  });

  /**
   * A click on the text answers, a drag across it marks. The two are told
   * apart by what the mouse left behind: a drag leaves a selection, and with a
   * tool active the release has already been spent on the mark.
   */
  const clickText = (event: React.MouseEvent<HTMLElement>) => {
    if (highlightMode !== 'none') return;
    const selection = window.getSelection();
    if (selection && !selection.isCollapsed) return;
    // A click on an existing mark is aimed at the mark (it opens the Remove
    // popover), not at the answer.
    if ((event.target as HTMLElement).closest?.(`[${RC_HIGHLIGHT_ATTR}]`)) return;
    pick();
  };

  return (
    <li
      className={cn(
        // rc-choice is a hook for the accessibility schemes, which need to
        // neutralise these fills. It carries no styling of its own.
        'rc-choice group relative rounded-lg border transition-all duration-200',
        'grid grid-cols-[auto_1fr_auto] items-start gap-3 p-3',
        isCorrect
          ? 'rc-choice-correct'
          : selected
            ? 'border-slate-500 bg-slate-100 shadow-sm'
            : eliminated
              ? 'border-transparent opacity-55 grayscale-[0.5]'
              : 'border-transparent hover:border-slate-200 hover:bg-slate-50',
      )}
    >
      <button
        type="button"
        onClick={pick}
        disabled={disabled || eliminated}
        aria-label={`Choice ${choice.letter}`}
        aria-describedby={textId}
        className={cn(
          'rc-choice-badge mt-0.5 flex h-6 w-6 shrink-0 items-center justify-center rounded-full border font-semibold transition-all',
          isCorrect
            ? ''
            : selected
              ? 'rc-choice-badge-selected border-slate-900 bg-slate-900 text-white'
              : 'border-slate-300 group-hover:border-slate-400',
        )}
        style={badgeStyle}
      >
        {choice.letter}
      </button>

      {/* Keyed on the question so a new question mounts fresh text nodes.
          Painting splits the nodes React rendered, and React would patch the
          first fragment in place and leave the rest behind. */}
      <div
        key={questionId}
        id={textId}
        ref={textRef}
        data-rc-mark-surface
        onMouseUp={surface.onMouseUp}
        onClick={clickText}
        className={cn(
          'rc-choice-text relative text-left leading-relaxed transition-colors',
          pickable && highlightMode === 'none' ? 'cursor-pointer' : 'cursor-text',
          eliminated && 'line-through',
        )}
        style={textStyle}
      >
        <RcMarkdown>{choice.text}</RcMarkdown>
      </div>

      <button
        type="button"
        onClick={(e) => {
          e.stopPropagation();
          // Guarded here as well as on the element: a disabled attribute alone
          // still leaves the handler reachable by a synthetic event.
          if (!noCrossOut) onToggleElimination(choice.letter);
        }}
        disabled={noCrossOut}
        aria-label={
          eliminated ? `Restore choice ${choice.letter}` : `Cross out choice ${choice.letter}`
        }
        title={eliminated ? 'Restore' : 'Cross out'}
        className={cn(
          'mt-0.5 p-1 transition-opacity',
          eliminated
            ? 'text-slate-700'
            : 'text-slate-400 opacity-0 hover:text-slate-800 focus-visible:opacity-100 group-hover:opacity-100',
        )}
      >
        {eliminated ? <EyeOff size={16} /> : <Eye size={16} />}
      </button>

      {children && <div className="col-start-2 col-end-4">{children}</div>}
    </li>
  );
}

export default RcChoiceRow;

import * as React from 'react';
import type { RcSelectionAnchor } from '../components/RcSelectionPopover';
import { RC_HIGHLIGHT_ATTR, captureRenderedSelection } from '../lib/rcHighlightPaint';
import type { RcAnnotationKind, RcHighlightColor, RcHighlightMode } from '../lib/rcTypes';

/**
 * The mouse-up ladder for one markable surface: the passage, a question stem,
 * or an answer choice. Every surface runs the same ladder so the toolbar tools,
 * the eraser and the selection popover behave identically wherever the student
 * releases the mouse.
 *
 *   erase          -> drag clears the sweep, click clears the mark under it
 *   tool active    -> a selection is applied on release, no popover
 *   click on mark  -> "Remove highlight" popover
 *   selection      -> "Highlight" popover
 *
 * The popover itself belongs to the pane, since a pane with several surfaces
 * (a stem and five choices) shows one popover. `onAnchor` hands the pane the
 * position together with a handle back to the surface that asked, so the
 * popover's buttons act on the right text.
 */

export interface RcMarkSurfaceHandle {
  /** Popover "Highlight": paint the selection the last mouse-up captured. */
  applyPending: (chosen: RcHighlightColor) => void;
  /** Popover "Remove highlight": drop the mark the last mouse-up landed on. */
  removePending: () => void;
}

export interface RcMarkSurfaceOptions {
  /** The element whose rendered text the offsets are measured against. */
  textRef: React.RefObject<HTMLElement | null>;
  /** The positioned, scrolling ancestor the popover is anchored inside. */
  paneRef: React.RefObject<HTMLElement | null>;
  highlightMode: RcHighlightMode;
  color: RcHighlightColor;
  onAdd: (start: number, end: number, color: RcHighlightColor, kind: RcAnnotationKind) => void;
  onRemove: (offset: number) => void;
  onRemoveRange: (start: number, end: number) => void;
  onAnchor: (anchor: RcSelectionAnchor | null, surface: RcMarkSurfaceHandle) => void;
}

/** Rendered-text offset of the first character inside a painted mark. */
function offsetOfMark(el: HTMLElement, mark: Element): number {
  const before = document.createRange();
  before.selectNodeContents(el);
  before.setEnd(mark, 0);
  return before.toString().length;
}

export function useRcMarkSurface(options: RcMarkSurfaceOptions) {
  // The handle is created once and read by the pane's popover later, so it
  // reads the options through a ref rather than closing over one render's.
  const latest = React.useRef(options);
  latest.current = options;

  const pending = React.useRef<{ start: number; end: number } | null>(null);
  const removeOffset = React.useRef<number | null>(null);

  const handle = React.useMemo<RcMarkSurfaceHandle>(() => {
    const self: RcMarkSurfaceHandle = {
      applyPending: (chosen) => {
        if (pending.current) {
          latest.current.onAdd(pending.current.start, pending.current.end, chosen, 'highlight');
        }
        window.getSelection()?.removeAllRanges();
        latest.current.onAnchor(null, self);
      },
      removePending: () => {
        if (removeOffset.current != null) latest.current.onRemove(removeOffset.current);
        latest.current.onAnchor(null, self);
      },
    };
    return self;
  }, []);

  /**

    // fire. Treating it as a release on the text would dismiss the popover
    // first, and the button would be gone before its click arrived.

    // With the eraser active, a click clears the mark under it and a drag
    // clears everything the sweep touches. No popover either way.
    if (highlightMode === 'erase') {
      if (selection) {
        onRemoveRange(selection.start, selection.end);
        window.getSelection()?.removeAllRanges();
      } else if (mark) {
        onRemove(offsetOfMark(el, mark));
      }
      onAnchor(null, handle);
      return;
    }

    // A marker or the underline is active: apply on release, like the official
    // interface, rather than asking again through the popover.
    if (highlightMode !== 'none' && selection) {
      const kind: RcAnnotationKind = highlightMode === 'underline' ? 'underline' : 'highlight';
      onAdd(
        selection.start,
        selection.end,
        kind === 'underline' ? color : (highlightMode as RcHighlightColor),
        kind,
      );
      window.getSelection()?.removeAllRanges();
      onAnchor(null, handle);
      return;
    }

    if (!selection && mark) {
      // Clicking an existing highlight offers to remove it.
      removeOffset.current = offsetOfMark(el, mark);
      onAnchor(anchorFor(mark.getBoundingClientRect(), 'remove'), handle);
      return;
    }

    if (!selection) {
      onAnchor(null, handle);
      return;
    }

    pending.current = { start: selection.start, end: selection.end };
    const rect = window.getSelection()?.getRangeAt(0)?.getBoundingClientRect();
    if (rect) onAnchor(anchorFor(rect, 'apply'), handle);
  };

  return { onMouseUp, ...handle };
}

/**
 * Painting highlights and find-text matches onto already-rendered markdown.
 *
 * Highlights are stored as character offsets, never as DOM nodes, so they
 * survive a reload and can be restored on any future encounter with a passage.
 * The offsets are measured against the *rendered* text of the container rather
 * than the raw markdown source: markdown syntax (the asterisks around
 * **Passage A**) never appears on screen, so an offset into the raw string does
 * not line up with anything the student can select. Rendering is deterministic
 * from the same source, so rendered offsets round-trip reliably.
 *
 * Painting walks text nodes and wraps the overlapping slice of each one, which
 * lets a highlight span element boundaries (a range crossing a bold run, say)
 * without needing the range to be well-formed for surroundContents.
 */

import { rcHighlightColor, type RcHighlight } from './rcTypes';

export const RC_HIGHLIGHT_ATTR = 'data-rc-highlight';
export const RC_FIND_ATTR = 'data-rc-find';

export interface RcTextRange {
  start: number;
  end: number;
  color?: string;
  /** 'highlight' or 'underline'; read by the caller's styleFor. */
  kind?: string;
}

/** Every text node under `root`, in document order. */
function textNodes(root: HTMLElement): Text[] {
  const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
  const nodes: Text[] = [];
  let node = walker.nextNode();
  while (node) {
    nodes.push(node as Text);
    node = walker.nextNode();
  }
  return nodes;
}

/**
 * The current selection expressed as offsets into `container`'s rendered text.
 * Returns null when there is no selection, it is collapsed, or it falls outside.
 */
export function captureRenderedSelection(
  container: HTMLElement,
): { start: number; end: number; text: string } | null {
  const selection = window.getSelection();
  if (!selection || selection.rangeCount === 0 || selection.isCollapsed) return null;

  const range = selection.getRangeAt(0);
  if (!container.contains(range.commonAncestorContainer)) return null;

  const text = selection.toString();
  if (!text.trim()) return null;

  const before = document.createRange();
  before.selectNodeContents(container);
  before.setEnd(range.startContainer, range.startOffset);
  const start = before.toString().length;

  return { start, end: start + text.length, text };
}

/** Removes marks previously painted with the given attribute, restoring the text. */
export function clearMarks(container: HTMLElement, attr: string): void {
  const marks = Array.from(container.querySelectorAll(`mark[${attr}]`));
  for (const mark of marks) {
    const parent = mark.parentNode;
    if (!parent) continue;
    while (mark.firstChild) parent.insertBefore(mark.firstChild, mark);
    parent.removeChild(mark);
  }
  container.normalize();
}

/**
 * Wraps each range in a <mark>. Ranges are in rendered-text coordinates.
 * Existing marks with the same attribute are cleared first, so this is
 * idempotent and safe to call on every render.
 */
export function paintRanges(
  container: HTMLElement,
  ranges: RcTextRange[],
  attr: string,
  styleFor?: (range: RcTextRange) => Partial<CSSStyleDeclaration>,
): void {
  clearMarks(container, attr);
  if (ranges.length === 0) return;

  for (const range of ranges) {
    if (range.end <= range.start) continue;
    // Re-walk per range: painting mutates the tree, so offsets must be
    // recomputed against the current DOM each time.
    let offset = 0;
    for (const node of textNodes(container)) {
      const length = node.data.length;
      const nodeStart = offset;
      const nodeEnd = offset + length;
      offset = nodeEnd;

      if (nodeEnd <= range.start || nodeStart >= range.end) continue;

      const from = Math.max(0, range.start - nodeStart);
      const to = Math.min(length, range.end - nodeStart);
      if (to <= from) continue;

      // Split the node so the middle piece is exactly the overlap.
      const middle = from > 0 ? node.splitText(from) : node;
      if (to - from < middle.data.length) middle.splitText(to - from);

      const mark = document.createElement('mark');
      mark.setAttribute(attr, range.color ?? 'true');
      const style = styleFor?.(range);
      if (style) Object.assign(mark.style, style);
      middle.parentNode?.insertBefore(mark, middle);
      mark.appendChild(middle);
    }
  }
}

/**
 * Paints a student's stored marks onto one surface: the passage, a stem, or a
 * choice. One styling for all of them, so a mark looks the same wherever it
 * sits. Callers pass only the marks that belong to this surface.
 */
export function paintRcAnnotations(container: HTMLElement, highlights: RcHighlight[]): void {
  paintRanges(
    container,
    highlights.map((h) => ({
      start: h.start,
      end: h.end,
      color: h.color,
      kind: h.kind ?? 'highlight',
    })),
    RC_HIGHLIGHT_ATTR,
    (r) =>
      r.kind === 'underline'
        ? {
            backgroundColor: 'transparent',
            textDecoration: 'underline',
            textDecorationColor: 'currentColor',
            textDecorationThickness: '2px',
            textUnderlineOffset: '3px',
          }
        : {
            backgroundColor: rcHighlightColor(r.color),
            // Multiply keeps the text readable through the wash, and matches
            // how the LR player renders its markers.
            mixBlendMode: 'multiply',
          },
  );
}

/** Case-insensitive matches of `query` in the container's rendered text. */
export function findMatches(container: HTMLElement, query: string): RcTextRange[] {
  if (query.trim().length < 2) return [];
  const haystack = (container.textContent ?? '').toLowerCase();
  const needle = query.toLowerCase();
  const matches: RcTextRange[] = [];
  let from = 0;
  for (;;) {
    const at = haystack.indexOf(needle, from);
    if (at === -1) break;
    matches.push({ start: at, end: at + needle.length });
    from = at + needle.length;
  }
  return matches;
}

/** Scrolls the nth painted find-match into view. */
export function scrollToMatch(container: HTMLElement, index: number): void {
  const marks = container.querySelectorAll(`mark[${RC_FIND_ATTR}]`);
  const target = marks[index] as HTMLElement | undefined;
  target?.scrollIntoView({ block: 'center', behavior: 'smooth' });
  marks.forEach((m, i) => m.classList.toggle('rc-find-active', i === index));
}

/** Dual tags are one string joined with " / ". Split on exactly that. */
export function splitTag(tag: string | null): string[] {
  if (!tag) return [];
  return tag
    .split(' / ')
    .map((t) => t.trim())
    .filter(Boolean);
}

/**
 * A choice with no tag and no explanation gets no row at all, even when a
 * response percentage exists: a row carrying only "1% of students" is the empty
 * shell the design rules out. The student's own pick always gets a row so they
 * can find it.
 */
export function isDisclosureEmpty(choice: RcChoice, isPicked: boolean): boolean {
  if (isPicked) return false;
  return !choice.tag && !choice.explanation;
