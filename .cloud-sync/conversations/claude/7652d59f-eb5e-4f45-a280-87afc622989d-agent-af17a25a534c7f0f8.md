export type HighlightColor = 'yellow' | 'pink' | 'orange' | 'underline';

/** What the toolbar's marker buttons can be set to. */
export type HighlightMode = HighlightColor | 'none' | 'erase';

/**
 * Where a highlight lives. Each section is its own raw string with its own
 * 0-based offsets: the stimulus, the stem, and every answer choice by letter.
 * Offsets from one section never mean anything in another, which is why every
 * overlap test below is scoped to the section first.
 */
export type HighlightSection = 'stimulus' | 'stem' | `choice:${string}`;

export function choiceSection(key: string): HighlightSection {
  return `choice:${key}`;
}

export interface Highlight {
  id: string;
  start: number;
  end: number;
  text: string;
  color: HighlightColor;
  section: HighlightSection;
}

/**
 * A highlightable block on screen: the element that holds one section's text
 * and the raw string it was rendered from. The source string is what offsets
 * are measured against, so `text` on a stored highlight is always a slice of
 * it rather than whatever the DOM happened to serialise.
 */
export interface HighlightRegion {
  section: HighlightSection;
  container: HTMLElement;
  source: string;
}

export interface CapturedSelection {
  section: HighlightSection;
  start: number;
  end: number;
  text: string;
}

/**
 * Clamp the live selection to `container`. Returns null when the selection is
 * collapsed or does not touch the container at all. A selection that starts
 * before the container (on the question number, in the previous block) or
 * runs past its end is cut at the container's edges rather than rejected,
 * so a drag that overshoots still highlights the part it covered.
 */
function selectionWithin(container: HTMLElement): Range | null {
  const selection = window.getSelection();
  if (!selection || selection.rangeCount === 0 || selection.isCollapsed) return null;

  const raw = selection.getRangeAt(0);
  const startInside = container.contains(raw.startContainer);
  const endInside = container.contains(raw.endContainer);
  if (!startInside && !endInside) {
    // Either the whole container sits inside the selection or none of it does.
    const touches =
      typeof raw.intersectsNode === 'function'
        ? raw.intersectsNode(container)
        : raw.commonAncestorContainer.contains(container);
    if (!touches) return null;
  }

  const range = raw.cloneRange();
  if (!startInside) range.setStart(container, 0);
  if (!endInside) range.setEnd(container, container.childNodes.length);
  if (range.collapsed) return null;
  return range;
}

export function captureTextSelection(container: HTMLElement): { start: number; end: number; text: string } | null {
  const range = selectionWithin(container);
  if (!range) return null;

  const selectedText = range.toString();
  if (!selectedText.trim()) return null;

  // ── Accurate offset mapping via data-para-start annotations ──────────────
  // HighlightedText annotates each <p> / dialogue-turn body with its raw-string
  // start offset. Within a single annotated segment there are no block boundaries,
  // so DOM text length == raw text length — no \n\n drift.
  const segments = Array.from(container.querySelectorAll<HTMLElement>('[data-para-start]'));

  if (segments.length > 0) {
    const segStart = (seg: HTMLElement) => parseInt(seg.getAttribute('data-para-start') ?? '0');

    // Returns the character offset from the start of `seg` to `node:nodeOffset`
    const withinOffset = (seg: HTMLElement, node: Node, nodeOffset: number): number => {
      const r = document.createRange();
      r.setStart(seg, 0);
      r.setEnd(node, nodeOffset);
      return r.toString().length;
    };

    // Find the annotated segment that contains a given DOM node
    const findSeg = (node: Node): HTMLElement | null => {
      for (const seg of segments) {
        if (seg === node || seg.contains(node)) return seg;
      }
      return null;
    };

    // A boundary that landed outside every segment (on the question number,
    // in padding, at the clamped container edge) snaps to the nearest segment
    // edge inside the selection instead of falling back to a DOM-length count
    // that would shift the whole highlight by the stray characters.
    const segmentRange = (seg: HTMLElement) => {
      const r = document.createRange();
      r.selectNodeContents(seg);
      return r;
    };
    const firstSegmentAfterStart = (): HTMLElement | null =>
      segments.find((seg) => range.compareBoundaryPoints(Range.START_TO_START, segmentRange(seg)) <= 0) ?? null;
    const lastSegmentBeforeEnd = (): HTMLElement | null => {
      for (let i = segments.length - 1; i >= 0; i--) {
        if (range.compareBoundaryPoints(Range.END_TO_END, segmentRange(segments[i])) >= 0) return segments[i];
      }
      return null;
    };

    const startSeg = findSeg(range.startContainer);
    const endSeg = findSeg(range.endContainer);

    let rawStart: number | null = null;
    let rawEnd: number | null = null;
    if (startSeg) {
      rawStart = segStart(startSeg) + withinOffset(startSeg, range.startContainer, range.startOffset);
    } else {
      const seg = firstSegmentAfterStart();
      if (seg) rawStart = segStart(seg);
    }
    if (endSeg) {
      rawEnd = segStart(endSeg) + withinOffset(endSeg, range.endContainer, range.endOffset);
    } else {
      const seg = lastSegmentBeforeEnd();
      if (seg) rawEnd = segStart(seg) + (seg.textContent?.length ?? 0);
    }

    if (rawStart !== null && rawEnd !== null && rawEnd > rawStart) {
      return { start: rawStart, end: rawEnd, text: selectedText };
    }
    if (rawStart !== null && rawEnd !== null) return null;
  }

  // ── Fallback for containers without data-para-start ──────────────────────
  const containerRange = document.createRange();
  containerRange.selectNodeContents(container);
  const beforeRange = containerRange.cloneRange();
  beforeRange.setEnd(range.startContainer, range.startOffset);
  const start = beforeRange.toString().length;
  const end   = start + selectedText.length;
  return { start, end, text: selectedText };
}

/**
 * One drag can cross the stem and two answer choices. Every region it touches
 * gets its own capture, each clipped to that region's text, so the student
 * gets exactly what she swept rather than only the block the mouse came up in.
 * Whitespace-only slivers (the gap between blocks) are dropped.
 */
export function captureSelectionAcross(regions: HighlightRegion[]): CapturedSelection[] {
  const captured: CapturedSelection[] = [];
  for (const region of regions) {
    const hit = captureTextSelection(region.container);
    if (!hit) continue;
    const start = Math.max(0, Math.min(hit.start, region.source.length));
    const end = Math.max(start, Math.min(hit.end, region.source.length));
    const text = region.source.slice(start, end);
    if (!text.trim()) continue;
    captured.push({ section: region.section, start, end, text });
  }
  return captured;
}

/**
 * Add a highlight, cutting away whatever it overlaps in the same section
 * ("last action wins"). Highlights in other sections are untouched: the
 * stem's offset 0 and the stimulus's offset 0 are different characters.
 */
export function replaceOverlappingHighlights(
  existing: Highlight[],
  newHighlight: Highlight
): Highlight[] {
  const result: Highlight[] = [];

  for (const h of existing) {
    // Different section, or no overlap - keep as is
    if (h.section !== newHighlight.section || h.end <= newHighlight.start || h.start >= newHighlight.end) {
      result.push(h);
      continue;
    }

    // Has overlap - split and keep non-overlapping portions
    // Keep the part before the new highlight
    if (h.start < newHighlight.start) {
      const beforeText = h.text.slice(0, newHighlight.start - h.start);
      result.push({
        ...h,
        id: `${h.id}-before`,
        end: newHighlight.start,
        text: beforeText
      });
    }

    // Keep the part after the new highlight
    if (h.end > newHighlight.end) {
      const offset = newHighlight.end - h.start;
      const afterText = h.text.slice(offset);
      result.push({
        ...h,
        id: `${h.id}-after`,
        start: newHighlight.end,
        text: afterText
      });
    }
  }

  // Add the new highlight
  result.push(newHighlight);

  // Merge adjacent highlights of the same color
  return mergeAdjacentSameColor(result);
}

/**

    if (h.section !== section || h.end <= start || h.start >= end) {
      result.push(h);
      continue;
    }

      result.push({ ...h, id: `${h.id}-before`, end: start, text: h.text.slice(0, start - h.start) });
    }
    if (h.end > end) {
      result.push({ ...h, id: `${h.id}-after`, start: end, text: h.text.slice(end - h.start) });
    }
  }
  return result;
}

function mergeAdjacentSameColor(highlights: Highlight[]): Highlight[] {
  if (highlights.length <= 1) return highlights;

  const sorted = [...highlights].sort((a, b) =>
    a.section === b.section ? a.start - b.start : a.section < b.section ? -1 : 1,
  );
  const merged: Highlight[] = [];
  let current = sorted[0];

  for (let i = 1; i < sorted.length; i++) {
    const next = sorted[i];

    // If adjacent and same color, merge
    if (next.start === current.end && next.color === current.color && next.section === current.section) {
      current = {
        ...current,
        end: next.end,
        text: current.text + next.text,
      };
    } else {
      merged.push(current);
      current = next;
    }
  }
  merged.push(current);

  return merged;
}

export function mergeOverlappingHighlights(highlights: Highlight[]): Highlight[] {
  if (highlights.length <= 1) return highlights;

  const sorted = [...highlights].sort((a, b) => a.start - b.start);
  const merged: Highlight[] = [];
  let current = sorted[0];

  for (let i = 1; i < sorted.length; i++) {
    const next = sorted[i];

    // If overlapping or adjacent, merge and prefer the latest color
    if (next.start <= current.end) {
      current = {
        ...next, // Use next highlight's properties (newer)
        start: Math.min(current.start, next.start),
        end: Math.max(current.end, next.end),
      };
    } else {
      merged.push(current);
      current = next;
    }
  }
  merged.push(current);

  return merged;
}

/**
 * Per-question highlight state for any surface that shows an LR question with
 * the marker toolbar: the drill, blind review, the section walk, the bootcamp
 * capstone modal. One implementation, so a highlight behaves the same in all
 * of them. Nothing here is persisted; highlights live as long as the surface.
 *
 * `applySelection` takes every region the pane offers and highlights the part
 * of the live selection that falls in each, so one drag across the stem and
 * two choices marks all three.

export function useQuestionHighlights() {
  const [highlights, setHighlights] = React.useState<Map<string, Highlight[]>>(new Map());
  const [history, setHistory] = React.useState<Map<string, Highlight[]>[]>([]);

  const remember = React.useCallback((snapshot: Map<string, Highlight[]>) => {
    setHistory((prev) => [...prev, new Map(snapshot)].slice(-MAX_HISTORY));
  }, []);

  const highlightsFor = React.useCallback(
    (qid: string | undefined): Highlight[] => (qid ? highlights.get(qid) ?? [] : []),
    [highlights],
  );

  /**
   * Turn the live selection into highlights (or erasures) on `qid`.
   * Returns true when something changed, which is also when the selection has
   * been cleared; a caller can use that to tell a marking drag from a click.
   */
  const applySelection = React.useCallback(
    (qid: string | undefined, mode: HighlightMode, regions: HighlightRegion[]): boolean => {
      if (!qid || mode === 'none') return false;
      const captured = captureSelectionAcross(regions);
      if (captured.length === 0) return false;

      let next = highlights.get(qid) ?? [];
      for (const hit of captured) {
        if (mode === 'erase') {
          next = eraseRange(next, hit.section, hit.start, hit.end);
        } else {
          next = replaceOverlappingHighlights(next, {
            id: crypto.randomUUID(),
            start: hit.start,
            end: hit.end,
            text: hit.text,
            color: mode as HighlightColor,
            section: hit.section,
          });
        }
      }

      remember(highlights);
      setHighlights(new Map(highlights).set(qid, next));
      window.getSelection()?.removeAllRanges();
      return true;
    },
    [highlights, remember],
  );

  /** The eraser's click on a highlight: the whole thing goes. */
  const removeHighlight = React.useCallback(
    (qid: string | undefined, highlightId: string) => {
      if (!qid) return;
      const current = highlights.get(qid) ?? [];
      if (!current.some((h) => h.id === highlightId)) return;
      remember(highlights);
      setHighlights(new Map(highlights).set(qid, current.filter((h) => h.id !== highlightId)));
    },
    [highlights, remember],
  );

  const undo = React.useCallback(() => {
    if (history.length === 0) return;
    setHighlights(new Map(history[history.length - 1]));
    setHistory(history.slice(0, -1));
  }, [history]);

  const clearHistory = React.useCallback(() => setHistory([]), []);

  return {
    highlights,
    highlightsFor,
    applySelection,
    removeHighlight,
    undo,
    canUndo: history.length > 0,
    clearHistory,
  };
}

// ─── Search-term overlay for plain-text segments only ───────────────────────
// Wraps case-insensitive matches inside a <mark> WITHOUT touching user
// highlight spans. Returns React nodes for the given plain text slice.
function renderWithSearch(text: string, searchTerm: string | undefined, keyPrefix: string): React.ReactNode {
  if (!searchTerm || searchTerm.trim().length < 2) return text;
  const term = searchTerm.trim();
  const lower = text.toLowerCase();
  const lowerTerm = term.toLowerCase();
  if (!lower.includes(lowerTerm)) return text;

  const nodes: React.ReactNode[] = [];
  let cursor = 0;
  let i = 0;
  while (cursor < text.length) {
    const idx = lower.indexOf(lowerTerm, cursor);
    if (idx === -1) {
      nodes.push(text.slice(cursor));
      break;
    }
    if (idx > cursor) nodes.push(text.slice(cursor, idx));
    nodes.push(
      <mark
        key={`${keyPrefix}-s${i++}`}
        style={{ backgroundColor: 'rgba(56, 189, 248, 0.55)', color: 'inherit', padding: 0 }}
      >
        {text.slice(idx, idx + term.length)}
      </mark>,
    );
    cursor = idx + term.length;
  }
  return <>{nodes}</>;
}

// ─── Inline segment renderer ─────────────────────────────────────────────────
function renderSegments(
  text: string,
  highlights: Highlight[],
  prefix: string,
  eraserMode: boolean | undefined,
  onHighlightClick: ((id: string) => void) | undefined,
  searchTerm?: string,
): React.ReactNode {
  if (highlights.length === 0) {
    return renderWithSearch(text, searchTerm, prefix);
  }

  const sorted = [...highlights].sort((a, b) => a.start - b.start);
  const nodes: React.ReactNode[] = [];
  let cursor = 0;

  for (let i = 0; i < sorted.length; i++) {
    const h = sorted[i];
    if (h.start > cursor) {
      const slice = text.slice(cursor, h.start);
      nodes.push(<span key={`${prefix}-t${i}`}>{renderWithSearch(slice, searchTerm, `${prefix}-t${i}`)}</span>);
    }
    nodes.push(
      <span
        key={`${prefix}-h${h.id}`}
        data-highlight-id={h.id}
        className={cn('hl', 'transition-all', eraserMode && 'cursor-pointer hover:opacity-60')}
        style={getHighlightStyles(h.color)}
        onClick={() => eraserMode && onHighlightClick?.(h.id)}
      >
        {text.slice(h.start, h.end)}
      </span>
    );
    cursor = h.end;
  }

  if (cursor < text.length) {
    const slice = text.slice(cursor);
    nodes.push(<span key={`${prefix}-tend`}>{renderWithSearch(slice, searchTerm, `${prefix}-tend`)}</span>);
  }

  return <>{nodes}</>;
}

  const dialogueTurns = rawSegments
    .map(seg => {
      const match = seg.match(speakerPattern);
      if (match && looksLikeSpeakerLabel(match[1])) {
        return { speaker: match[1], text: seg.slice(match[0].length).trim() };
      }
      return null;
    })
    .filter(Boolean);

    dialogueTurns.length >= 2 &&
    dialogueTurns.length === rawSegments.length &&
    dialogueTurns.every(t => t !== null);

    const turnRanges = dialogueTurns.map(turn => {
      const speakerPrefix = `${turn!.speaker}: `;
      const turnStart = text.indexOf(speakerPrefix, currentOffset);
      const start = turnStart + speakerPrefix.length;
      const end = start + turn!.text.length;
      currentOffset = end;
      return { turn: turn!, start, end };
    });

      <div className="space-y-4">
        {turnRanges.map((tr, idx) => {
          const turnHighlights = highlights
            .filter(h => h.start < tr.end && h.end > tr.start)
            .map(h => ({
              ...h,
              start: Math.max(0, h.start - tr.start),
              end: Math.min(tr.turn.text.length, h.end - tr.start),
            }));

          return (
            <div key={idx} className="leading-relaxed" style={{ lineHeight: 1.7 }}>
              <span className="font-bold">{tr.turn.speaker}:</span>{' '}
              <span className="inline" data-para-start={tr.start}>
                {renderSegments(tr.turn.text, turnHighlights, `d${idx}`, eraserMode, onHighlightClick, searchTerm)}
              </span>
            </div>
          );
        })}
      </div>
    );
  }

  const paragraphs = text.split('\n\n');

  let currentOffset = 0;
  const paragraphRanges = paragraphs.map(para => {
    const start = currentOffset;
    const end = currentOffset + para.length;
    currentOffset = end + 2;
    return { start, end, text: para };
  });

  return (
    <>
      {paragraphRanges.map((pr, idx) => {
        const paraHighlights = highlights
          .filter(h => h.start < pr.end && h.end > pr.start)
          .map(h => ({
            ...h,

          </p>
        );
      })}
    </>
  );
}

/**
 * The mouse-up ladder for one markable surface: the passage, a question stem,
 * or an answer choice. Every surface runs the same ladder so the toolbar tools,
 * the eraser and the selection popover behave identically wherever the student
 * releases the mouse.
 *
 *   erase          -> drag clears the sweep, click clears the mark under it
 *   tool active    -> a selection is applied on release, no popover
 *   click on mark  -> "Remove highlight" popover
 *   selection      -> "Highlight" popover
 *
 * The popover itself belongs to the pane, since a pane with several surfaces
 * (a stem and five choices) shows one popover. `onAnchor` hands the pane the
 * position together with a handle back to the surface that asked, so the
 * popover's buttons act on the right text.
 */

export interface RcMarkSurfaceHandle {
  /** Popover "Highlight": paint the selection the last mouse-up captured. */
  applyPending: (chosen: RcHighlightColor) => void;
  /** Popover "Remove highlight": drop the mark the last mouse-up landed on. */
  removePending: () => void;
}

/**
 * Painting highlights and find-text matches onto already-rendered markdown.
 *
 * Highlights are stored as character offsets, never as DOM nodes, so they
 * survive a reload and can be restored on any future encounter with a passage.
 * The offsets are measured against the *rendered* text of the container rather
 * than the raw markdown source: markdown syntax (the asterisks around
 * **Passage A**) never appears on screen, so an offset into the raw string does
 * not line up with anything the student can select. Rendering is deterministic
 * from the same source, so rendered offsets round-trip reliably.
 *
 * Painting walks text nodes and wraps the overlapping slice of each one, which
 * lets a highlight span element boundaries (a range crossing a bold run, say)
 * without needing the range to be well-formed for surroundContents.
 */

import { rcHighlightColor, type RcHighlight } from './rcTypes';

export const RC_HIGHLIGHT_ATTR = 'data-rc-highlight';
export const RC_FIND_ATTR = 'data-rc-find';

export interface RcTextRange {
  start: number;
  end: number;
  color?: string;
  /** 'highlight' or 'underline'; read by the caller's styleFor. */
  kind?: string;
}

/** Every text node under `root`, in document order. */
function textNodes(root: HTMLElement): Text[] {
  const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
  const nodes: Text[] = [];
  let node = walker.nextNode();
  while (node) {
    nodes.push(node as Text);
    node = walker.nextNode();
  }
  return nodes;
}

/**
 * The current selection expressed as offsets into `container`'s rendered text.
 * Returns null when there is no selection, it is collapsed, or it falls outside.
 */
export function captureRenderedSelection(
  container: HTMLElement,
): { start: number; end: number; text: string } | null {
  const selection = window.getSelection();
  if (!selection || selection.rangeCount === 0 || selection.isCollapsed) return null;

  const range = selection.getRangeAt(0);
  if (!container.contains(range.commonAncestorContainer)) return null;

  const text = selection.toString();
  if (!text.trim()) return null;

  const before = document.createRange();
  before.selectNodeContents(container);
  before.setEnd(range.startContainer, range.startOffset);
  const start = before.toString().length;

  return { start, end: start + text.length, text };
}

/** Removes marks previously painted with the given attribute, restoring the text. */
export function clearMarks(container: HTMLElement, attr: string): void {
  const marks = Array.from(container.querySelectorAll(`mark[${attr}]`));
  for (const mark of marks) {
    const parent = mark.parentNode;
    if (!parent) continue;
    while (mark.firstChild) parent.insertBefore(mark.firstChild, mark);
    parent.removeChild(mark);
  }
  container.normalize();
}

/**
 * Wraps each range in a <mark>. Ranges are in rendered-text coordinates.
 * Existing marks with the same attribute are cleared first, so this is
 * idempotent and safe to call on every render.
 */
export function paintRanges(
  container: HTMLElement,
  ranges: RcTextRange[],
  attr: string,
  styleFor?: (range: RcTextRange) => Partial<CSSStyleDeclaration>,
): void {
  clearMarks(container, attr);
  if (ranges.length === 0) return;

  for (const range of ranges) {
    if (range.end <= range.start) continue;
    // Re-walk per range: painting mutates the tree, so offsets must be
    // recomputed against the current DOM each time.
    let offset = 0;
    for (const node of textNodes(container)) {
      const length = node.data.length;
      const nodeStart = offset;
      const nodeEnd = offset + length;
      offset = nodeEnd;

      if (nodeEnd <= range.start || nodeStart >= range.end) continue;

      const from = Math.max(0, range.start - nodeStart);
      const to = Math.min(length, range.end - nodeStart);
      if (to <= from) continue;

      // Split the node so the middle piece is exactly the overlap.
      const middle = from > 0 ? node.splitText(from) : node;
      if (to - from < middle.data.length) middle.splitText(to - from);

      const mark = document.createElement('mark');
      mark.setAttribute(attr, range.color ?? 'true');
      const style = styleFor?.(range);
      if (style) Object.assign(mark.style, style);
      middle.parentNode?.insertBefore(mark, middle);
      mark.appendChild(middle);
    }
  }
}

/** Case-insensitive matches of `query` in the container's rendered text. */
export function findMatches(container: HTMLElement, query: string): RcTextRange[] {
  if (query.trim().length < 2) return [];
  const haystack = (container.textContent ?? '').toLowerCase();
  const needle = query.toLowerCase();
  const matches: RcTextRange[] = [];
  let from = 0;
  for (;;) {
    const at = haystack.indexOf(needle, from);
    if (at === -1) break;
    matches.push({ start: at, end: at + needle.length });
    from = at + needle.length;
  }
  return matches;
}

  const target = marks[index] as HTMLElement | undefined;
  target?.scrollIntoView({ block: 'center', behavior: 'smooth' });
  marks.forEach((m, i) => m.classList.toggle('rc-find-active', i === index));
}

 * two drills are indistinguishable at the choice level.
 *
 * Two deliberate departures from LR, both taken from the official RC interface:
 * the letter badge is a circle with a visible outline rather than a rounded
 * square with a transparent border, and the eliminate control is always visible
 * rather than appearing on row hover.
 *
 * Like LR, a revealed question greens the credited answer and leaves the
 * student's pick in the selected slate — there is no red row. The verdict line
 * and the "Your answer" chip carry that.
 *
 * The credited answer's colours come from `.rc-choice-correct` in rc.css rather
 * than Tailwind utilities. They used to be green-500/green-50 here while the
 * verdict banner directly above used the design system's drill-correct #2f9e64
 * on #eaf6ef — two greens for one verdict, on one screen. Both read the same
 * token now, and it is the documented one.
 *
 * The choice text is a markable surface, like the passage: the student can
 * highlight and underline inside it with the same tools. That is why the text
 * is not a button. Chromium will not start a drag-selection inside one, and a
 * disabled button swallows the mouse entirely, which would make a revealed or
 * crossed-out choice impossible to mark. A plain click on the text still
 * answers; the letter badge stays the keyboard and screen-reader control.
