/**
 * Minutes on the clock for a timing mode, or null when there is no clock.
 * Standard is 35; 52.5 and 70 are the 1.5x and 2x accommodations.
 */
export function sectionMinutes(mode: TimerMode | undefined): number | null {
  switch (mode) {
    case '52.5':
      return 52.5;
    case '70':
      return 70;
    case 'unlimited':
      return null;
    default:
      return 35;
