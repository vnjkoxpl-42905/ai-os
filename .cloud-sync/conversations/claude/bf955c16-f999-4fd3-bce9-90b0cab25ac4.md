66	- Whitespace collapsed to single spaces; em dash → "-"; **no paragraphs**.
67	- Sections: sentences (regex `[^.!?]+([.!?"”]+|$)`) accumulated until ~2000 chars; a sentence that would push past 2400 is split at a word boundary. Sentence ends after "Dr/Mr/Mrs/Ms" or a single letter do not close a section.
68	- Words = split(" "); id 1-based; charPosition.
69	- Progress % = round((sectionIndex + posInSection/sectionLen) / sections * 100) (1 when on the last word of a section).
70	- Remaining = time from current word to end: "N hour(s) and M minute(s)", "Less than a minute".

72	### Layout (Serial / Flow / Highlight / Scroll)
73	- Words are measured in the font; a line takes words while `x + width + space <= stageWidth - 40` **and** words < Words Per Line.

let memory: Db | null = null;

export function readDb(): Db {
  if (memory) return memory;
  try {
    const raw = localStorage.getItem(KEY);
    if (raw) {
      const parsed = JSON.parse(raw) as Db;
      memory = { ...seedDb(), ...parsed, preferences: { ...DEFAULT_PREFERENCES, ...parsed.preferences } };
      return memory;
    }
  } catch {
    // Blocked or corrupt storage: fall through to a fresh in-memory copy.
  }
  memory = seedDb();
  writeDb(memory);
  return memory;
}

375	        }}

380	        <EReaderSearch open={searchOpen} onClose={() => { setSearchOpen(false); setMarks([]); }} words={model.words} pageOf={pageOf} onSelect={(p, hits) => { setPage(p); setMarks(hits); }} />

384	        title={passage.title}
385	        onOkay={() => {
386	          saveProgress.mutate({ passage_id: passage.id, word_index: 0, completed: true });
387	          if (document.fullscreenElement) document.exitFullscreen();
388	          navigate(RC_LIBRARY_PATH);
389	        }}
390	      />
391	    </div>
392	  );
393	}
