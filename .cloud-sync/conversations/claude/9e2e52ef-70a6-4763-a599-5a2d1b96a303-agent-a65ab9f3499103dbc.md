112	  const fallback = { index: 0, step: 0, answers: {} as Record<string, Answer>, archived: [] as Archived[] };
113	  try {

115	    if (!data || typeof data !== 'object') return fallback;
116	    const archived: Archived[] = Array.isArray(data.archived) ? data.archived.slice(-3) : [];
117	    if (!data.answers || typeof data.answers !== 'object') return { ...fallback, archived };
118	    const current = data.fingerprint === itemsFingerprint(items) || data.fingerprint === JSON.stringify(items);
119	    if (!current) return Object.keys(data.answers).length ? { ...fallback, archived: [...archived, { fingerprint: data.fingerprint, index: data.index, step: data.step, answers: data.answers }].slice(-3) } : { ...fallback, archived };
120	    const index = Number.isInteger(data.index) && data.index >= 0 && data.index < items.length ? data.index : 0;
121	    const step = Number.isInteger(data.step) && data.step >= 0 && data.step < items[index].stages.length ? data.step : 0;
122	    const answers: Record<string, Answer> = {};
123	    for (const [key, value] of Object.entries(data.answers)) {

182	function practiceDiagrams(item: PracticeItem) {
183	  const left = `${item.leftNegated ? '~' : ''}${item.leftCode}`;
184	  const right = `${item.rightNegated ? '~' : ''}${item.rightCode}`;
185	  const notLeft = `${item.leftNegated ? '' : '~'}${item.leftCode}`;
186	  const notRight = `${item.rightNegated ? '' : '~'}${item.rightCode}`;
187	  return { original: `${left} → ${right}`, contra: `${notRight} → ${notLeft}`, reverse: `${right} → ${left}`, inverse: `${notLeft} → ${notRight}` };
188	}
189	function practiceChoices(item: PracticeItem, contrapositive: boolean) {
190	  const diagrams = practiceDiagrams(item);
191	  const correct = contrapositive ? diagrams.contra : diagrams.original;
192	  const wrong = [diagrams.reverse, diagrams.inverse];
193	  const position = contrapositive ? item.contraPosition : item.originalPosition;
194	  const options = [...wrong];
195	  options.splice(position, 0, correct);
196	  return options;
197	}

/** True once the element has scrolled into view. Without IntersectionObserver (old browsers, tests) it is true straight away. */
function useInView<T extends Element>(): [React.RefObject<T>, boolean] {
  const ref = useRef<T>(null);
  const [inView, setInView] = useState(typeof IntersectionObserver === 'undefined');
  useEffect(() => {
    const node = ref.current;
    if (!node || typeof IntersectionObserver === 'undefined') return;
    const observer = new IntersectionObserver(entries => { if (entries.some(entry => entry.isIntersecting)) { setInView(true); observer.disconnect(); } }, { threshold: 0.6 });
    observer.observe(node);
    return () => observer.disconnect();
  }, []);
  return [ref, inView];
}
