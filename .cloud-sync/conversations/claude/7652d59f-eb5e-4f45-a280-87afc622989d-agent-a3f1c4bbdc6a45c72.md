import * as React from 'react';
import { loadAnnotations, saveAnnotations } from '../api/rcProgress';
import type {
  RcAnnotationKind,
  RcHighlight,
  RcHighlightColor,
  RcHighlightTarget,
} from '../lib/rcTypes';

/**
 * Per-passage highlights, loaded once for the session's passages and saved back
 * (debounced) as the student paints. Keyed by passage so they restore on any
 * future encounter with that passage, in this session or a later one.
 */
export function useRcHighlights(passageIds: number[]) {
  const [byPassage, setByPassage] = React.useState<Map<number, RcHighlight[]>>(new Map());
  const [color, setColor] = React.useState<RcHighlightColor>('yellow');
  const [loaded, setLoaded] = React.useState(false);
  /**
   * Mirrors `byPassage`, but updated synchronously on every write. Reading the
   * state variable inside the mutators would capture whatever the last render
   * closed over, so two marks made between renders — which is simply someone
   * highlighting quickly — would compute from the same base and the first
   * would be lost.
   */
  const latest = React.useRef<Map<number, RcHighlight[]>>(new Map());

  const key = passageIds.join(',');
  React.useEffect(() => {
    let cancelled = false;
    if (passageIds.length === 0) return;
    loadAnnotations(passageIds)
      .then((map) => {
        if (!cancelled) {
          latest.current = map;
          setByPassage(map);
          setLoaded(true);
        }
      })
      .catch(() => {
        // Highlights are an aid, not the drill. A load failure must not block it.
        if (!cancelled) setLoaded(true);
      });
    return () => {
      cancelled = true;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [key]);

  const forPassage = React.useCallback(
    (passageId: number) => byPassage.get(passageId) ?? [],
    [byPassage],
  );

  const write = React.useCallback((passageId: number, next: RcHighlight[]) => {
    const map = new Map(latest.current);
    map.set(passageId, next);
    latest.current = map;
    setByPassage(map);
    saveAnnotations(passageId, next);
  }, []);

  const add = React.useCallback(
    (
      passageId: number,
      target: RcHighlightTarget,
      start: number,
      end: number,
      /** Passed in by the popover so picking a swatch applies that color now. */
      chosen: RcHighlightColor = color,
      kind: RcAnnotationKind = 'highlight',
    ) => {
      if (end <= start) return;
      const existing = latest.current.get(passageId) ?? [];
      // Same target + same kind is one layer. A highlight and an underline can
      // sit on the same words, so they never interact.
      const sameLayer = (h: RcHighlight) =>
        h.target === target && (h.kind ?? 'highlight') === kind;

      // Marking over the same color extends one continuous span rather than
      // leaving a seam. Repeat until nothing new is absorbed: swallowing one
      // neighbour can bring the next into reach.
      let lo = start;
      let hi = end;
      for (let grew = true; grew; ) {
        grew = false;
        for (const h of existing) {
          if (!sameLayer(h) || h.color !== chosen) continue;
          if (h.end < lo || h.start > hi) continue;
          if (h.start < lo || h.end > hi) {
            lo = Math.min(lo, h.start);
            hi = Math.max(hi, h.end);
            grew = true;
          }
        }
      }

      // Marking over a different color trims the old span around the new one
      // instead of discarding it, so highlighting three words inside a
      // sentence keeps the rest of that sentence.
      const next: RcHighlight[] = [];
      for (const h of existing) {
        if (!sameLayer(h)) {
          next.push(h);
          continue;
        }
        if (h.color === chosen && h.end >= lo && h.start <= hi) continue;
        if (h.end <= lo || h.start >= hi) {
          next.push(h);
          continue;
        }
        if (h.start < lo) next.push({ ...h, end: lo });
        if (h.end > hi) next.push({ ...h, start: hi });
      }
      next.push({ target, start: lo, end: hi, color: chosen, kind });
      write(passageId, next);
    },
    [color, write],
  );

  const removeAt = React.useCallback(
    (passageId: number, target: RcHighlightTarget, offset: number) => {
      const existing = latest.current.get(passageId) ?? [];
      write(
        passageId,
        existing.filter((h) => !(h.target === target && offset >= h.start && offset < h.end)),
      );
    },
    [write],
  );

  /** Eraser drag: clear every annotation the swept range touches. */
  const removeRange = React.useCallback(
    (passageId: number, target: RcHighlightTarget, start: number, end: number) => {
      const existing = latest.current.get(passageId) ?? [];
      write(
        passageId,
        existing.filter((h) => h.target !== target || h.end <= start || h.start >= end),
      );
    },
    [write],
  );

  return { forPassage, add, removeAt, removeRange, color, setColor, loaded };
}

 * Painting highlights and find-text matches onto already-rendered markdown.
 *
 * Highlights are stored as character offsets, never as DOM nodes, so they
 * survive a reload and can be restored on any future encounter with a passage.
 * The offsets are measured against the *rendered* text of the container rather
 * than the raw markdown source: markdown syntax (the asterisks around
 * **Passage A**) never appears on screen, so an offset into the raw string does
 * not line up with anything the student can select. Rendering is deterministic
 * from the same source, so rendered offsets round-trip reliably.
 *
 * Painting walks text nodes and wraps the overlapping slice of each one, which
 * lets a highlight span element boundaries (a range crossing a bold run, say)
 * without needing the range to be well-formed for surroundContents.
 */

import { rcHighlightColor, type RcHighlight } from './rcTypes';

export const RC_HIGHLIGHT_ATTR = 'data-rc-highlight';
export const RC_FIND_ATTR = 'data-rc-find';

export interface RcTextRange {
  start: number;
  end: number;
  color?: string;
  /** 'highlight' or 'underline'; read by the caller's styleFor. */
  kind?: string;
}

/** Every text node under `root`, in document order. */
function textNodes(root: HTMLElement): Text[] {
  const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
  const nodes: Text[] = [];
  let node = walker.nextNode();
  while (node) {
    nodes.push(node as Text);
    node = walker.nextNode();
  }
  return nodes;
}

/**
 * The current selection expressed as offsets into `container`'s rendered text.
 * Returns null when there is no selection, it is collapsed, or it falls outside.
 */
export function captureRenderedSelection(
  container: HTMLElement,
): { start: number; end: number; text: string } | null {
  const selection = window.getSelection();
  if (!selection || selection.rangeCount === 0 || selection.isCollapsed) return null;

  const range = selection.getRangeAt(0);
  if (!container.contains(range.commonAncestorContainer)) return null;

  const text = selection.toString();
  if (!text.trim()) return null;

  const before = document.createRange();
  before.selectNodeContents(container);
  before.setEnd(range.startContainer, range.startOffset);
  const start = before.toString().length;

  return { start, end: start + text.length, text };
}

/** Removes marks previously painted with the given attribute, restoring the text. */
export function clearMarks(container: HTMLElement, attr: string): void {
  const marks = Array.from(container.querySelectorAll(`mark[${attr}]`));
  for (const mark of marks) {
    const parent = mark.parentNode;
    if (!parent) continue;
    while (mark.firstChild) parent.insertBefore(mark.firstChild, mark);
    parent.removeChild(mark);
  }
  container.normalize();
}

/**
 * Wraps each range in a <mark>. Ranges are in rendered-text coordinates.
 * Existing marks with the same attribute are cleared first, so this is
 * idempotent and safe to call on every render.
 */
export function paintRanges(
  container: HTMLElement,
  ranges: RcTextRange[],
  attr: string,
  styleFor?: (range: RcTextRange) => Partial<CSSStyleDeclaration>,
): void {
  clearMarks(container, attr);
  if (ranges.length === 0) return;

  for (const range of ranges) {
    if (range.end <= range.start) continue;
    // Re-walk per range: painting mutates the tree, so offsets must be
    // recomputed against the current DOM each time.
    let offset = 0;
    for (const node of textNodes(container)) {
      const length = node.data.length;
      const nodeStart = offset;
      const nodeEnd = offset + length;
      offset = nodeEnd;

      if (nodeEnd <= range.start || nodeStart >= range.end) continue;

      const from = Math.max(0, range.start - nodeStart);
      const to = Math.min(length, range.end - nodeStart);
      if (to <= from) continue;

      // Split the node so the middle piece is exactly the overlap.
      const middle = from > 0 ? node.splitText(from) : node;
      if (to - from < middle.data.length) middle.splitText(to - from);

      const mark = document.createElement('mark');
      mark.setAttribute(attr, range.color ?? 'true');
      const style = styleFor?.(range);
      if (style) Object.assign(mark.style, style);
      middle.parentNode?.insertBefore(mark, middle);
      mark.appendChild(middle);
    }
  }
            // how the LR player renders its markers.
            mixBlendMode: 'multiply',
          },
  );
}

/** Case-insensitive matches of `query` in the container's rendered text. */
export function findMatches(container: HTMLElement, query: string): RcTextRange[] {
  if (query.trim().length < 2) return [];
  const haystack = (container.textContent ?? '').toLowerCase();
  const needle = query.toLowerCase();
  const matches: RcTextRange[] = [];
  let from = 0;

 * right column is present at all (RC's Passage Only View, and the LR
 * pool-exhausted state, both render a single full-width pane).
 *
 * Both drill runners currently pass `passage-weighted` with no page cap, so the
 * defaults below are for other callers rather than for them. The doc used to

export type DrillSplitRatio = 'even' | 'passage-weighted';

const RATIO_CLASS: Record<DrillSplitRatio, string> = {

};

  left: React.ReactNode;
  /** Omit for a single full-width column. */
  right?: React.ReactNode;
  ratio?: DrillSplitRatio;
  /** Tailwind cap for the page width, or null to fill the frame. */
  maxWidthClassName?: string | null;
  className?: string;
}

  left,
  right,
  ratio = 'even',
  maxWidthClassName = 'max-w-7xl',
  className,
}: Props) {

        right ? RATIO_CLASS[ratio] : 'grid-cols-1',

      {left}
      {right}

  );
