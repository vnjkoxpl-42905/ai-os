/**
 * The clock a full section was started under, in milliseconds. Unlimited has
 * no budget, which is not the same as a budget of zero.
 */
function sectionBudgetMs(timer?: TimerConfig): number | undefined {
  if (!timer) return undefined;
  switch (timer.mode) {
    case '35':
      return 35 * 60 * 1000;
    case '52.5':
      return 52.5 * 60 * 1000;
    case '70':
      return 70 * 60 * 1000;
    case 'custom':
      return timer.customMinutes ? timer.customMinutes * 60 * 1000 : undefined;
    default:
      return undefined;
  }
}

/**
 * A homework set built from one real section — every qid comes from the same
 * PT and section number, the shape `assign.ts` and the admin set builder both
 * produce today. Such a set is a section wearing a homework row, not a mixed
 * question list, so it runs under a real countdown instead of the untimed
 * stopwatch every other homework set gets. A set that mixes sections (or PTs)
 * returns null and keeps the untimed behaviour, because there is no one clock
 * that honestly times a blend.
 */
export function singleSectionOf(qids: string[]): { pt: number; section: number } | null {
  if (qids.length === 0) return null;
  let pt: number | null = null;
  let section: number | null = null;
  for (const qid of qids) {
    const match = QID_PATTERN.exec(qid);
    if (!match) return null;
    const qPt = Number(match[1]);
    const qSection = Number(match[2]);
    if (pt === null) {
      pt = qPt;
      section = qSection;
    } else if (qPt !== pt || qSection !== section) {
      return null;
    }
  }
  return pt !== null && section !== null ? { pt, section } : null;
}

function parseLegacy(p: URLSearchParams): DrillLaunch | null {
  const mode = p.get('mode');
  if (!mode) return null;

  const returnTo = safeReturnTo(p.get('returnTo'));
  const sealed = p.get('sealed') === '1';
  const initialIndex = nonNegativeInt(p.get('initialIndex'));

  if (mode === 'full-section') {

    const section = positiveNumber(p.get('section'));
    if (pt && section) {

        kind: 'section',

        // 52.5, 70 and unlimited are the accommodated clocks. Reducing them to
        // a number of minutes would quietly re-time an accommodated sitting.

        minutes: positiveNumber(p.get('minutes')),
        sealed: sealed || undefined,
        ptLeg: readPtLeg(p.get('runId'), p.get('leg')),
        options: undefinedIfEmpty({
          ...(returnTo ? { returnTo } : {}),
          ...(legacyLabel(p) ? { label: legacyLabel(p) as string } : {}),
          ...(initialIndex != null ? { initialIndex } : {}),
        }),
      };
    }
    // Falls through exactly as before: a malformed section link became an empty
    // type-drill rather than an error.
  }

  return {
    kind: 'questions',
    qids: list(p.get('qids')),
    options: undefinedIfEmpty({
      ...(returnTo ? { returnTo } : {}),
      ...(legacyLabel(p) ? { label: legacyLabel(p) as string } : {}),
      ...(initialIndex != null ? { initialIndex } : {}),
      ...(p.get('present') === 'smart' ? { present: 'smart' as const } : {}),
    }),
    carry: readLegacyCarry(
      p,
      (mode === 'practice-set' ? 'type-drill' : mode) as DrillMode,
    ),
  };
}

export function buildSectionStats(
  session: DrillSession,
  brResults?: BlindReviewResult[],
): SectionQuestionStat[] {
  // Who was already answered at the buzzer. A section that never ran out has
  // no mark, and no question on it is in overtime.
  const atBuzzer = session.regulationMark
    ? new Set(session.regulationMark.answeredQids)
    : null;
  const visited = new Set(session.visitedQids ?? []);

  return session.questionQueue
    .map((qid, idx) => {
      const question = questionBank.getQuestion(qid);
      const attempt = session.attempts.get(qid);
      const brResult = brResults?.find((br) => br.qid === qid);

      if (!question) return null;

      // No attempt means the question was never answered, which counts as wrong.
      return {
        qid,
        qnum: idx + 1,
        qtype: question.qtype,
        difficulty: question.difficulty,
        initialAnswer: attempt?.selectedAnswer || UNANSWERED,
        correctAnswer: question.correctAnswer,
        initialCorrect: attempt?.correct || false,
        brAnswer: brResult?.brAnswer,
        brCorrect: brResult?.correct,
        brChanged: brResult?.brChanged,
        timeMs: attempt?.timeMs || 0,
        brTimeMs: brResult?.brTimeMs,
        switchCount: attempt?.switchCount || 0,
        reasoningTypes: question.reasoningType?.length ? question.reasoningType : undefined,
        questionFeatures: question.questionFeatures?.length
          ? question.questionFeatures
          : undefined,
        answerTrail: attempt?.answerTrail,
        visitCount: attempt?.visitCount,
        noChangeRevisits: attempt?.noChangeRevisits,
        eliminations: attempt?.eliminations,
        // Undefined rather than false when nothing was recorded at all: a
        // sitting from before visits were tracked must not report every
        // question as one the student never opened.
        everVisited: visited.size > 0 ? visited.has(qid) : undefined,
        answeredInOvertime: atBuzzer ? !atBuzzer.has(qid) : undefined,
        flagged: attempt?.brMarked || false,
      };
    })
    .filter(Boolean) as SectionQuestionStat[];
}
