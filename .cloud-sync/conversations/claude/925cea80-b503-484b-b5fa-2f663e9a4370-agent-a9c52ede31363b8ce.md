// Task -> action wiring for the student to-do list.
//
// A weekly_plan_item may carry an `action` object in metadata_json describing
// what tapping the task should launch and what surface completing it maps to:
//   { kind: 'drill'|'homework'|'bootcamp'|'plain', ref_id?: string, route?: string }
// 'plain' (or absent) means a free-text checkbox with no launch target.

export type TodoActionKind = 'drill' | 'homework' | 'bootcamp' | 'plain';

export interface TodoAction {
  kind: TodoActionKind;
  ref_id?: string;
  route?: string;
}

/** Read the action object off an item's metadata_json, defaulting to plain. */
export function readTodoAction(metadata: Record<string, unknown> | null | undefined): TodoAction {
  const raw = (metadata as { action?: unknown } | null | undefined)?.action;
  if (!raw || typeof raw !== 'object') return { kind: 'plain' };
  const a = raw as Record<string, unknown>;
  const kind = a.kind;
  if (kind !== 'drill' && kind !== 'homework' && kind !== 'bootcamp') {
    return { kind: 'plain' };
  }
  return {
    kind,
    ref_id: typeof a.ref_id === 'string' ? a.ref_id : undefined,
    route: typeof a.route === 'string' ? a.route : undefined,
  };
}

/**
 * Resolve a task action to the student-facing route to launch, or null when the
 * task is plain (nothing to open). An explicit `route` wins; otherwise we derive
 * a sensible path per kind from `ref_id`, mirroring resolveItemRoute in
 * src/lib/classroomItems.ts.
 */
export function resolveTodoRoute(action: TodoAction, returnTo = '/schedule'): string | null {
  if (action.kind === 'plain') return null;
  if (action.route) return action.route;
  const ref = action.ref_id ? encodeURIComponent(action.ref_id) : null;
  switch (action.kind) {
    case 'drill':
      return ref
        ? `/drill?assignmentId=${ref}&returnTo=${encodeURIComponent(returnTo)}`
        : '/drill';
    case 'homework':
      return ref ? `/classroom/${ref}` : '/classroom';
    case 'bootcamp':
      return ref ? `/bootcamp/${ref}` : '/bootcamps';
    default:
      return null;
  }
}

/**
 * `/guided/:id` reads `returnTo` to decide where finishing the drill lands the
 * student. Building it into the row would bake in whichever surface happened to
 * construct the feed, which is how every guided drill ended up returning to
 * `/schedule` no matter where it was launched from. Consumers call this at
 * click time with their own location instead.
 */
export function routeWithReturnTo(route: string, from: string): string {
  if (!route.startsWith('/guided/')) return route;
  const [path, query = ''] = route.split('?');
  const params = new URLSearchParams(query);
  params.set('returnTo', from);
  return `${path}?${params.toString()}`;
}

        kind: 'ta',
        refId: a.id,
        sourceTable: 'ta_assignments',

        dueAt: null,

    // so they collapse to one. An open assignment always wins over a completed
    // one — otherwise finishing the older twin would hide work still owed —
    // and the newest wins within that.
    const bestByDrill = new Map<string, GuidedRow>();
    for (const g of guided) {
      const drillId = g.guided_drill_id ?? g.id;
      const held = bestByDrill.get(drillId);
      if (!held) {
        bestByDrill.set(drillId, g);
        continue;
      }
      const heldOpen = held.status !== 'completed';
      const thisOpen = g.status !== 'completed';
      if (thisOpen !== heldOpen) {
        if (thisOpen) bestByDrill.set(drillId, g);
        continue;
      }
      if (new Date(g.assigned_at).getTime() > new Date(held.assigned_at).getTime()) {
        bestByDrill.set(drillId, g);
      }
    }
    for (const g of bestByDrill.values()) {
      put({
        key: `g-${g.id}`,
        kind: 'guided',
        refId: g.id,
        sourceTable: 'guided_drill_assignments',
        title: g.guided_drill?.title || 'Guided drill',
        subtitle: `${g.guided_drill?.question_count ?? '?'} questions${g.guided_drill?.estimated_minutes ? ` · ${g.guided_drill.estimated_minutes} min` : ''}`,
        status: g.status === 'completed' ? 'completed' : g.status === 'assigned' ? 'assigned' : 'in_progress',
        createdAt: g.assigned_at,
        dueAt: null,
        route: `/guided/${encodeURIComponent(g.id)}`,
      });
    }

    for (const a of activities) {
      put({
        key: `act-${a.id}`,
        kind: 'activity',
        refId: a.id,
        sourceTable: 'activity_assignments',
        title: a.title || 'Activity',
        subtitle: a.subtitle || 'Activity',
        status: normalizeStatus(a.status),
        createdAt: a.created_at,
        dueAt: null,
        route: `/activities/${encodeURIComponent(a.id)}`,
      });
    }

    for (const item of items) {
      const kind = WORK_ITEM_KIND[item.type];
      if (!kind) continue; // notes and recaps are reading material, not work
      const route = resolveItemRoute(item);
      if (!route) continue;
      const sourceTable = item.source_table ?? 'classroom_items';
      const refId = item.source_id ?? item.id;
      const existing = byIdentity.get(`${sourceTable}:${refId}`);
      // Completion wins over the mirror. A student who finished the underlying
      // assignment must not see it come back because the classroom_items row it
      // was mirrored into still says "assigned" — that is why finished work kept
      // reappearing on Home.
      const itemStatus = normalizeStatus(item.status);
      const status =
        itemStatus === 'completed' || existing?.status === 'completed' ? 'completed' : itemStatus;
      byIdentity.set(`${sourceTable}:${refId}`, {
        key: `ci-${item.id}`,
        kind,
        refId,
        sourceTable,
        title: item.title || existing?.title || 'Assigned work',
        // The item carries no question count, so keep the source row's subtitle
        // when there is one.
        subtitle: existing?.subtitle ?? item.description ?? '',
        status,
        createdAt: item.created_at,
        dueAt: item.due_at,
        // The mirror row was written before the section rule existed, so the
        // source row's directions route wins over the item's classroom link.
        route: existing?.selfCompleting ? existing.route : route,
        selfCompleting:
          (item.metadata_json as { kind?: string } | null)?.kind === 'timed_section' ||
          !!existing?.selfCompleting,
        briefing: readSectionBriefing(item.metadata_json) ?? undefined,
      });
    }

    // Open work first, then completed; newest within each bucket.
    return [...byIdentity.values()].sort((x, y) => {
      const s = (x.status === 'completed' ? 1 : 0) - (y.status === 'completed' ? 1 : 0);
      if (s !== 0) return s;
      return new Date(y.createdAt).getTime() - new Date(x.createdAt).getTime();
    });
  }
}

const STATUS_SOURCES = new Set([
  'homework_assignments',
  'ta_assignments',
  'guided_drill_assignments',
  'activity_assignments',
  'weekly_plan_items',
]);

export async function completeAssignedWork(
  row: Pick<AssignedTodoRow, 'sourceTable' | 'refId'>,
  studentId: string,
): Promise<void> {
  const nowIso = new Date().toISOString();

  if (STATUS_SOURCES.has(row.sourceTable)) {
    const { error } = await supabase
      .from(row.sourceTable as 'homework_assignments')
      .update({ status: 'completed' })
      .eq('id', row.refId)
      .eq('student_id', studentId);
    // A source row the student cannot update (or that no longer exists) must
    // not block the classroom_items mirror below, which is what the lists read.
    if (error) console.warn('[completeAssignedWork] source update failed', error.message);
  }

  const mirror = supabase
    .from('classroom_items')
    .update({ status: 'completed', completed_at: nowIso })
    .eq('student_id', studentId)
    .neq('status', 'completed');

  const { error: mirrorError } =
    row.sourceTable === 'classroom_items'
      ? await mirror.eq('id', row.refId)
      : await mirror.eq('source_table', row.sourceTable).eq('source_id', row.refId);

  if (mirrorError) throw mirrorError;
}
