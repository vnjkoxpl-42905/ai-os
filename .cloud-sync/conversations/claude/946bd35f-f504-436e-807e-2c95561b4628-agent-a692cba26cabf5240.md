## Two machine-specific gotchas

- **rspack emits no JavaScript here.** `Config.setRspack(true)` (which
  `create-video` writes when you accept Tailwind) prints "Bundling 100%", exits
  0, and produces only `index.html`. The resulting `ENOENT ... bundle.js` is a
  decoy: it is the source-map reader failing, not the real fault. It is
  commented out in `remotion.config.ts`. Do not re-enable without testing a render.
- **npm blocks install scripts**, so esbuild's binary never installs. Fix with
  `npm approve-scripts esbuild`.

1. `align/transcribe.py` runs faster-whisper over the three MP3s for word-level

  const out = []; let cur = null;
  for (const raw of md.split(/\r?\n/)) {
    const line = raw.trim(); if (!line) continue;
    if (line.startsWith(">")) {
      const t = line.replace(/^>\s*/, "");
      if (!cur || cur.said) { cur = { read: [], said: "" }; out.push(cur); }
      cur.read.push(t);
    } else if (cur) cur.said += (cur.said ? " " : "") + line;
  }
  return out.filter(g => g.read.length);
}

/* Global Needleman-Wunsch over token sequences.
   Returns map[i] = index in B for each token i of A, or -1 when unaligned. */
function align(A, B) {
  const n = A.length, m = B.length;
  const M = 2, MIS = -1, GAP = -1;
  const eq = (a, b) => a === b || (a.length > 3 && b.length > 3 && a.slice(0, 4) === b.slice(0, 4));
  // score matrix as Int32Array rows
  const dp = new Int32Array((n + 1) * (m + 1));
  const at = (i, j) => i * (m + 1) + j;
  for (let i = 1; i <= n; i++) dp[at(i, 0)] = i * GAP;
  for (let j = 1; j <= m; j++) dp[at(0, j)] = j * GAP;
  for (let i = 1; i <= n; i++)
    for (let j = 1; j <= m; j++) {
      const d = dp[at(i - 1, j - 1)] + (eq(A[i - 1], B[j - 1]) ? M : MIS);
      const u = dp[at(i - 1, j)] + GAP, l = dp[at(i, j - 1)] + GAP;
      dp[at(i, j)] = d >= u ? (d >= l ? d : l) : (u >= l ? u : l);
    }
  const map = new Array(n).fill(-1);
  let i = n, j = m;
  while (i > 0 && j > 0) {
    const d = dp[at(i - 1, j - 1)] + (eq(A[i - 1], B[j - 1]) ? M : MIS);
    if (dp[at(i, j)] === d) { if (eq(A[i - 1], B[j - 1])) map[i - 1] = j - 1; i--; j--; }
    else if (dp[at(i, j)] === dp[at(i - 1, j)] + GAP) i--;
    else j--;
  }
  return map;
}
