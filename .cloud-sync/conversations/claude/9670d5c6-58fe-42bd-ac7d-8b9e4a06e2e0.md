178	function practiceDiagrams(item: PracticeItem) {
179	  const left = `${item.leftNegated ? '~' : ''}${item.leftCode}`;
180	  const right = `${item.rightNegated ? '~' : ''}${item.rightCode}`;
181	  const notLeft = `${item.leftNegated ? '' : '~'}${item.leftCode}`;
182	  const notRight = `${item.rightNegated ? '' : '~'}${item.rightCode}`;
183	  return { original: `${left} → ${right}`, contra: `${notRight} → ${notLeft}`, reverse: `${right} → ${left}`, inverse: `${notLeft} → ${notRight}` };

185	function practiceChoices(item: PracticeItem, contrapositive: boolean) {
186	  const diagrams = practiceDiagrams(item);
187	  const correct = contrapositive ? diagrams.contra : diagrams.original;
188	  const wrong = [diagrams.reverse, diagrams.inverse];
189	  const position = contrapositive ? item.contraPosition : item.originalPosition;
190	  const options = [...wrong];
191	  options.splice(position, 0, correct);
192	  return options;
193	}
